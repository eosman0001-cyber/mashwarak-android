function maybeStartFirstUseTour(){
  try{
    const params=new URLSearchParams(window.location.search||'');
    if(params.get('open'))return;
    if(localStorage.getItem(MASHWARAK_TOUR_KEY)==='1')return;
  }catch(e){}

  setTimeout(function(){
    startUsageTour(false);
  },800);
}

function startUsageTour(force){
  try{closeMainDrawer();}catch(e){}
  try{openScreen('customer');}catch(e){}

  usageTourActive=true;
  usageTourIndex=0;

  const overlay=document.getElementById('mwTourOverlay');
  if(overlay){
    overlay.classList.add('active');
    overlay.setAttribute('aria-hidden','false');
  }

  showUsageTourStep();
}

function showUsageTourStep(){
  if(!usageTourActive)return;

  const step=usageTourSteps[usageTourIndex];
  if(!step){
    finishUsageTour();
    return;
  }

  rideWizardGo(step.step);

  const title=document.getElementById('mwTourTitle');
  const text=document.getElementById('mwTourText');
  const count=document.getElementById('mwTourCounter');
  const back=document.getElementById('mwTourBack');
  const next=document.getElementById('mwTourNext');

  if(title)title.textContent=step.title;
  if(text)text.textContent=step.text;
  if(count)count.textContent=(usageTourIndex+1)+' من '+usageTourSteps.length;
  if(back)back.disabled=usageTourIndex===0;
  if(next)next.textContent=usageTourIndex===usageTourSteps.length-1?'تمام، فهمت':'التالي';

  setTimeout(function(){
    const target=document.querySelector(step.target);
    if(!target)return;

    target.scrollIntoView({behavior:'smooth',block:'center'});

    setTimeout(function(){
      updateUsageTourSpotlight();
    },350);
  },80);
}

function updateUsageTourSpotlight(){
  if(!usageTourActive)return;

  const step=usageTourSteps[usageTourIndex];
  const target=step?document.querySelector(step.target):null;
  const spot=document.getElementById('mwTourSpotlight');
  if(!target || !spot)return;

  const r=target.getBoundingClientRect();
  const pad=7;

  spot.style.top=Math.max(7,r.top-pad)+'px';
  spot.style.left=Math.max(7,r.left-pad)+'px';
  spot.style.width=Math.max(40,Math.min(window.innerWidth-14,r.width+pad*2))+'px';
  spot.style.height=Math.max(40,r.height+pad*2)+'px';
}

function usageTourNext(){
  if(usageTourIndex>=usageTourSteps.length-1){
    finishUsageTour();
    return;
  }
  usageTourIndex++;
  showUsageTourStep();
}

function usageTourPrev(){
  if(usageTourIndex<=0)return;
  usageTourIndex--;
  showUsageTourStep();
}

function finishUsageTour(){
  usageTourActive=false;

    overlay.setAttribute('aria-hidden','true');
  }

  try{
    localStorage.setItem(MASHWARAK_TOUR_KEY,'1');
  }catch(e){}

  rideWizardGo(1);
}

async function nativeCameraCaptured(input,target){
  if(!input || !input.files || !input.files[0]) return;

  try{
    const meta=captainUploadMeta(target);
    if(meta) document.getElementById(meta[1]).textContent='جاري تجهيز الصورة...';

    const packed=await compressCaptainImage(input.files[0]);
    captainFiles[target]=packed;

    if(meta){
      document.getElementById(meta[0]).classList.add('ready');
      document.getElementById(meta[1]).textContent='تم التصوير ✓';
    }

      document.getElementById(meta[0]).classList.remove('ready');

async function openLiveCamera(target){
  captainCameraTarget=target;
  const sheet=document.getElementById('captainCameraSheet');
  const video=document.getElementById('captainCameraVideo');
  document.getElementById('cameraTitle').textContent='تصوير '+(captainCameraLabels[target]||'المستند');
  const hint=document.getElementById('cameraHint');
  if(hint){
    hint.textContent = target==='liveSelfie'
      ? 'استخدم الكاميرا الأمامية | اجعل الوجه واضحًا وفي منتصف الإطار'
      : target==='captainWithCar'
        ? 'يجب أن يظهر الكابتن والسيارة بوضوح في نفس الصورة'
        : 'ضع المطلوب داخل الإطار بوضوح ثم اضغط زر التصوير';
  }

  lockBackgroundScroll('modal');
  sheet.classList.add('active');

  try{
    if(captainCameraStream){
      captainCameraStream.getTracks().forEach(t=>t.stop());
      captainCameraStream=null;
    }

    const facing = target==='liveSelfie' ? 'user' : 'environment';
    captainCameraStream=await navigator.mediaDevices.getUserMedia({
      video:{
        facingMode:{ideal:facing},
        width:{ideal:1920},
        height:{ideal:1080}
      },
      audio:false
    });
    video.srcObject=captainCameraStream;
    await video.play();
    return true;
  }catch(e){
    sheet.classList.remove('active');
    if(captainCameraStream){
      captainCameraStream.getTracks().forEach(t=>t.stop());
      captainCameraStream=null;
    }
    unlockBackgroundScroll();
    throw e;
  }
}

function closeLiveCamera(){
  const sheet=document.getElementById('captainCameraSheet');
  sheet.classList.remove('active');
  if(captainCameraStream){
    captainCameraStream.getTracks().forEach(t=>t.stop());
    captainCameraStream=null;
  }
  captainCameraTarget=null;
  unlockBackgroundScroll();
}

function captureLiveCamera(){
  if(!captainCameraTarget)return;
  const video=document.getElementById('captainCameraVideo');
  const canvas=document.getElementById('captainCameraCanvas');
  if(!video.videoWidth || !video.videoHeight){
    alert('الكاميرا لم تصبح جاهزة بعد.');
    return;
  }

  const maxSide=1400;
  let w=video.videoWidth,h=video.videoHeight;
  if(Math.max(w,h)>maxSide){
    const r=maxSide/Math.max(w,h);
    w=Math.round(w*r);h=Math.round(h*r);
  }

  canvas.width=w;canvas.height=h;
  const ctx=canvas.getContext('2d');
  ctx.drawImage(video,0,0,w,h);

  const dataUrl=canvas.toDataURL('image/jpeg',0.78);
  captainFiles[captainCameraTarget]={
    name:captainCameraTarget+'.jpg',
    mimeType:'image/jpeg',
    data:dataUrl.split(',')[1]
  };

  const meta=captainUploadMeta(captainCameraTarget);
  if(meta){
    document.getElementById(meta[0]).classList.add('ready');
    document.getElementById(meta[1]).textContent='تم التصوير ✓';
  }

  closeLiveCamera();
}

function receiptTextLtr(ctx,text,x,y,size,weight,color,align){
  ctx.save();
  ctx.direction='ltr';
  ctx.textAlign=align||'center';
  ctx.textBaseline='alphabetic';
  ctx.fillStyle=color||'#252832';
  ctx.font=(weight||'700')+' '+size+'px Cairo, Arial, sans-serif';
  ctx.fillText(String(text==null?'':text),x,y);
  ctx.restore();
}

function rr(ctx,x,y,w,h,r,fill,stroke){
  const radius=Math.min(r,w/2,h/2);
  ctx.beginPath();
  ctx.moveTo(x+radius,y);
  ctx.arcTo(x+w,y,x+w,y+h,radius);
  ctx.arcTo(x+w,y+h,x,y+h,radius);
  ctx.arcTo(x,y+h,x,y,radius);
  ctx.arcTo(x,y,x+w,y,radius);
  ctx.closePath();
  if(fill){ctx.fillStyle=fill;ctx.fill();}
  if(stroke){ctx.strokeStyle=stroke;ctx.lineWidth=2;ctx.stroke();}
}

function receiptText(ctx,text,x,y,size,weight,color,align){
  ctx.save();
  ctx.direction='rtl';
  ctx.textAlign=align||'right';
  ctx.textBaseline='alphabetic';
  ctx.fillStyle=color||'#252832';
  ctx.font=(weight||'700')+' '+size+'px Cairo, Arial, sans-serif';
  ctx.fillText(String(text==null?'':text),x,y);
  ctx.restore();
}

function receiptWrap(ctx,text,x,y,maxWidth,lineHeight,size,weight,color,maxLines){
  const words=String(text||'').split(/\s+/);
  let line='', lines=[], test='';
  ctx.save();
  ctx.font=(weight||'700')+' '+size+'px Cairo, Arial, sans-serif';
  for(const word of words){
    test=line?line+' '+word:word;
    if(ctx.measureText(test).width>maxWidth && line){
      lines.push(line);
      line=word;
      if(maxLines && lines.length>=maxLines)break;
    }else{
      line=test;
    }
  }
  if((!maxLines || lines.length<maxLines) && line)lines.push(line);
  ctx.restore();
  lines.forEach(function(l,i){
    receiptText(ctx,l,x,y+i*lineHeight,size,weight,color,'right');
  });
  return lines.length*lineHeight;
}

function openNativePicker(id){
  const el=document.getElementById(id);
  if(!el || el.disabled)return;

  try{
    // Keep the original tap activation alive. Blurring another control before
    // showPicker() can make mobile browsers ignore the request.
    if(document.activeElement && document.activeElement!==el &&
       /^(INPUT|TEXTAREA)$/.test(document.activeElement.tagName)){
      document.activeElement.blur();
    }

    try{el.focus({preventScroll:true});}
    catch(_){try{el.focus();}catch(__){}}

    if(typeof el.showPicker==='function'){
      el.showPicker();
      return;
    }

    // Older mobile browsers: a direct click on the native control is the best fallback.
    el.click();
  }catch(e){
    try{el.click();}catch(_){}
  }
}


function fill12HourSelects(prefix){
  const isReturn=prefix==='return';
  const h=document.getElementById(isReturn?'rideReturnTimeHour':'rideTimeHour');
  const m=document.getElementById(isReturn?'rideReturnTimeMinute':'rideTimeMinute');
  const p=document.getElementById(isReturn?'rideReturnTimePeriod':'rideTimePeriod');
  const hint=document.getElementById(isReturn?'rideReturnTimeHint':'rideTimeHint');
  if(!h||!m||!p)return;

  h.innerHTML='<option value="">ساعة</option>';
  for(let i=1;i<=12;i++){
    const o=document.createElement('option');
    o.value=String(i).padStart(2,'0');
    o.textContent=String(i).padStart(2,'0');
    h.appendChild(o);
  }

  m.innerHTML='<option value="">دقيقة</option>';
  for(let i=0;i<60;i+=5){
    const o=document.createElement('option');
    o.value=String(i).padStart(2,'0');
    o.textContent=String(i).padStart(2,'0');
    m.appendChild(o);
  }

  const now=new Date();
  let hour=now.getHours();
  const period=hour>=12?'PM':'AM';
  hour=hour%12||12;
  const minute=Math.round(now.getMinutes()/5)*5;
  const normalizedMinute=minute===60?55:minute;

  if(!isReturn){
    h.value=String(hour).padStart(2,'0');
    m.value=String(normalizedMinute).padStart(2,'0');
    p.value=period;
    sync12HourTime('ride');
  }else{
    h.value='';
    m.value='';
    p.value=period;
    sync12HourTime('return');
  }

  if(hint){
    const mm=String(now.getMinutes()).padStart(2,'0');
    hint.textContent='مثال حسب الوقت الحالي: '+String(hour).padStart(2,'0')+':'+mm+' '+period;
  }
}

function readFileAsDataURL(file){
  return new Promise(function(resolve,reject){
    const r=new FileReader();
    r.onload=function(){resolve(r.result)};
    r.onerror=reject;
    r.readAsDataURL(file);
  });
}

async function compressCaptainImage(file){
  if(!file || !file.type.startsWith('image/')) throw new Error('اختر صورة صحيحة.');
  if(file.size>12*1024*1024) throw new Error('حجم الصورة كبير جدًا. الحد الأقصى 12 MB قبل الضغط.');

  const src=await readFileAsDataURL(file);
  const img=await new Promise(function(resolve,reject){
    const im=new Image();
    im.onload=function(){resolve(im)};
    im.onerror=function(){reject(new Error('تعذر قراءة الصورة.'))};
    im.src=src;
  });

  const maxSide=1400;
  let w=img.width,h=img.height;
  if(Math.max(w,h)>maxSide){
    const ratio=maxSide/Math.max(w,h);
    w=Math.round(w*ratio);h=Math.round(h*ratio);
  }

  const canvas=document.createElement('canvas');
  canvas.width=w;canvas.height=h;
  const ctx=canvas.getContext('2d');
  ctx.drawImage(img,0,0,w,h);
  const dataUrl=canvas.toDataURL('image/jpeg',0.72);
  return {
    name:(file.name||'image').replace(/\.[^.]+$/,'')+'.jpg',
    mimeType:'image/jpeg',
    data:dataUrl.split(',')[1]
  };
}

let mwScrollLockCount=0;

function lockBackgroundScroll(mode){
  // Only full-screen overlays should lock the page.
  // Native <select> controls must never lock the main Google Sites iframe.
  if(mode==='select') return;

  mwScrollLockCount++;
  document.body.classList.add('mwModalOpen');
  document.documentElement.classList.add('mwModalOpen');
}

function unlockBackgroundScroll(force){
  if(force===true){
    mwScrollLockCount=0;
  }else{
    mwScrollLockCount=Math.max(0,mwScrollLockCount-1);
  }

  if(mwScrollLockCount===0){
    document.body.classList.remove('mwModalOpen','mwSelectOpen');
    document.documentElement.classList.remove('mwModalOpen');
    document.body.style.top='';
  }
}

function setAirportDirection(dir){
  airportDirection=dir==='FROM'?'FROM':'TO';
  document.getElementById('airportDirectionFrom').classList.toggle('active',airportDirection==='FROM');
  document.getElementById('airportDirectionTo').classList.toggle('active',airportDirection==='TO');

  const current=airportDirection==='FROM'?selectedFromAirport:selectedToAirport;
  const select=document.getElementById('airportSelect');
  if(select)select.value=current||'';
}

function openAirportSheet(dir){
  fillAirportSelect();
  setAirportDirection(dir||'TO');
  const sheet=document.getElementById('mwAirportSheet');
  if(!sheet)return;
  lockBackgroundScroll('modal');
  sheet.classList.add('active');
}

function closeAirportSheet(){
  const sheet=document.getElementById('mwAirportSheet');
  if(sheet)sheet.classList.remove('active');
  unlockBackgroundScroll(true);
}

function refreshAirportChosen(){
  const fromBox=document.getElementById('fromAirportChosen');
  const toBox=document.getElementById('toAirportChosen');

  if(fromBox){
    fromBox.classList.toggle('active',!!selectedFromAirport);
    fromBox.innerHTML=selectedFromAirport?'<b>المطار المختار:</b> '+selectedFromAirport:'';
  }
  if(toBox){
    toBox.classList.toggle('active',!!selectedToAirport);
    toBox.innerHTML=selectedToAirport?'<b>المطار المختار:</b> '+selectedToAirport:'';
  }
}

function confirmAirportSelection(){
  const airport=document.getElementById('airportSelect').value;
  if(!airport){
    alert('اختر المطار أولًا.');
    return;
  }

  if(airportDirection==='FROM'){
    selectedFromAirport=airport;
    const from=document.getElementById('rideFromCenter');
    from.value='AIRPORT';
    document.getElementById('rideFrom').value=airport;
    document.getElementById('fromVillageWrap').classList.remove('active','mwAttention');
    document.getElementById('fromOtherWrap').classList.remove('active');
  }else{
    selectedToAirport=airport;
    const to=document.getElementById('rideToGovernorate');
    to.value='AIRPORT';
    document.getElementById('rideTo').value=airport;
    document.getElementById('rideToDetail').value=airport;
    document.getElementById('toCenterWrap').classList.remove('active','mwAttention');
    document.getElementById('toVillageWrap').classList.remove('active','mwAttention');
  }

  refreshAirportChosen();
  closeAirportSheet();
  clearPreviousRideResultOnNewInput();
}

function syncRideFrom(){
  const center=document.getElementById('rideFromCenter').value;
  if(center==='AIRPORT'){
    document.getElementById('rideFrom').value=selectedFromAirport||'';
    refreshAirportChosen();
    return;
  }
  const village=document.getElementById('rideFromVillage').value;
  const other=document.getElementById('rideFromOther').value.trim();
  const area=village==='أخرى'?other:village;
  document.getElementById('rideFrom').value=center
    ? 'المنوفية - '+center+(area?' - '+area:'')
    : '';
}

function fillSelectOptions(select,placeholder,items){

  centerWrap.classList.toggle('mwAttention',!center);

function openRideContactSheet(){
  hideBox('rideError');

  const from=document.getElementById('rideFrom').value.trim();
  const to=document.getElementById('rideTo').value.trim();
  const gov=document.getElementById('rideToGovernorate').value;
  const toCenter=document.getElementById('rideToCenter').value;
  const toVillage=document.getElementById('rideToVillage').value;
  const date=document.getElementById('rideDate').value;
  const time=document.getElementById('rideTime').value;
  const missing=[];

  if(!from) missing.push(document.getElementById('rideFromCenter').value==='AIRPORT'?'اختيار مطار الانطلاق':'مكان الانطلاق');
  if(!gov) missing.push('المحافظة / الوجهة');
  if(gov && gov!=='AIRPORT' && !toCenter) missing.push('المركز / المدينة');
  if(gov && gov!=='AIRPORT' && !toVillage) missing.push('القرية / المنطقة');
  if(!to) missing.push('الوجهة');
  if(!date) missing.push('تاريخ المشوار');
  if(!time) missing.push('وقت المشوار');

  if(tripType==='ROUND'){
    if(!document.getElementById('rideReturnDate').value) missing.push('تاريخ العودة');
    if(!document.getElementById('rideReturnTime').value) missing.push('وقت العودة');
  }

  if(waitingRequired && !document.getElementById('rideWaitingDetails').value){
    missing.push('مدة الانتظار');
  }

  if(missing.length){
    showBox('rideError','يرجى استكمال البيانات التالية: '+missing.join(' | ')+'.');
    const err=document.getElementById('rideError');
    if(err) err.scrollIntoView({behavior:'smooth',block:'center'});
    return;
  }

  rideWizardStep=3;
  updateRideWizardUi();
  updateRideConfirmSummaryV2();

  dismissKeyboard();
  const sheet=document.getElementById('mwContactSheet');
  const card=sheet ? sheet.querySelector('.mwSheetCard') : null;
  if(card) card.scrollTop=0;

  lockBackgroundScroll('modal');
  if(sheet) sheet.classList.add('active');

  requestAnimationFrame(function(){
    if(card) card.scrollTop=0;
  });
}

function closeRideContactSheet(){
  const sheet=document.getElementById('mwContactSheet');
  const active=document.activeElement;
  if(active && sheet && sheet.contains(active) && typeof active.blur==='function'){
    try{active.blur();}catch(e){}
  }
  if(sheet)sheet.classList.remove('active');
  rideWizardStep=2;
  updateRideWizardUi();
  unlockBackgroundScroll(true);
}

function submitRide(){
  hideBox('rideSuccess');hideBox('rideError');
  const payload={
    customerName:document.getElementById('rideName').value,
    mobile:document.getElementById('rideMobile').value,
    tripType:tripType,
    from:document.getElementById('rideFrom').value,
    to:document.getElementById('rideTo').value,
    date:document.getElementById('rideDate').value,
    time:document.getElementById('rideTime').value,
    returnDate:document.getElementById('rideReturnDate').value,
    returnTime:document.getElementById('rideReturnTime').value,
    waiting:waitingRequired ? 'YES' : 'NO',
    waitingDetails:document.getElementById('rideWaitingDetails').value,
    termsAccepted:document.getElementById('rideTerms').checked,
    passengers:document.getElementById('ridePassengers').value,
    extras:selectedRideExtras(),
    captainAgePreference:captainAgePreference ? 'YES' : 'NO',
    captainAgeMin:document.getElementById('captainAgeMin').value,
    captainAgeMax:document.getElementById('captainAgeMax').value,
    acPreference:rideAcPreference,
    notes:''
  };

  if(!payload.acPreference){
    showBox('rideError','اختر هل تريد سيارة بتكييف شغال أم بدون تكييف.');
    return;
  }

  if(!payload.termsAccepted){
    showBox('rideError','يجب الموافقة على الشروط والأحكام قبل تأكيد الطلب.');
    return;
  }
  if(captainAgePreference){
    if(!payload.captainAgeMin || !payload.captainAgeMax){
      showBox('rideError','اختر نطاق سن الكابتن المفضل.');return;
    }
    if(Number(payload.captainAgeMin)>Number(payload.captainAgeMax)){
      showBox('rideError','سن البداية يجب أن يكون أقل من أو يساوي سن النهاية.');return;
    }
  }

  if(!payload.time){
    showBox('rideError','من فضلك حدد وقت المشوار أولاً.');
    return;
  }

  const submitBtn=document.getElementById('rideSubmitBtn');
  if(submitBtn){
    submitBtn.disabled=true;
    submitBtn.textContent='جاري إرسال الطلب...';
  }
  loading(true);
  google.script.run
    .withSuccessHandler(r=>{
      const submitBtn=document.getElementById('rideSubmitBtn');
      if(submitBtn){submitBtn.disabled=false;submitBtn.textContent='تأكيد طلب المشوار';}

      // Keep a clean snapshot for the downloadable receipt before clearing the form.
      lastRideReceipt=buildRideReceiptData(payload,r.rideId,r.estimatedPrice,r.estimateNote);

      closeRideContactSheet();
      unlockBackgroundScroll(true);
      resetRideForm(false);

      showBox('rideSuccess',(r.message||'تم استلام الطلب.')+' رقم الطلب: '+r.rideId);
      showRideEstimate(r);

      const dl=document.getElementById('downloadRideReceiptBtn');
      if(dl)dl.classList.add('show');

      const actions=document.getElementById('rideSuccessActions');
      if(actions)actions.classList.add('active');

      loading(false);
      setTimeout(function(){
        const success=document.getElementById('rideSuccess');
        if(success) success.scrollIntoView({behavior:'smooth',block:'center'});
      },80);
    })
    .withFailureHandler(e=>{
      loading(false);
      const submitBtn=document.getElementById('rideSubmitBtn');
      if(submitBtn){submitBtn.disabled=false;submitBtn.textContent='تأكيد طلب المشوار';}
      showBox('rideError',e.message||'تعذر إرسال الطلب');
    })
    .submitRideRequest(payload);
}

function statusLabel(s){
  return {NEW:'جديد',CONFIRMED:'مؤكد',ASSIGNED:'تم إسناد كابتن',COMPLETED:'تمت',CANCELLED:'ملغاة'}[s]||s;
}
function statusClass(s){
  return s==='NEW'?'b-new':s==='COMPLETED'?'b-ok':s==='CANCELLED'?'b-red':'b-blue';
}
function driverStatusLabel(s){
  return {UNDER_REVIEW:'تحت المراجعة',APPROVED:'معتمد',REJECTED:'مرفوض',INACTIVE:'غير نشط'}[s]||s;
}
function driverStatusClass(s){
  return s==='UNDER_REVIEW'?'b-new':s==='APPROVED'?'b-ok':s==='REJECTED'?'b-red':'b-blue';
}
function esc(s){
  return String(s==null?'':s).replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
}

            <div class="meta">${esc(p.serviceType)} • ${esc(p.price)} جنيه${waitText}</div>
          </div>
          <span class="badge ${p.active?'b-ok':'b-red'}">${p.active?'نشط':'موقوف'}</span>
        </div>
        <div class="card-actions">
          <button type="button" class="primary-mini" data-price-id="${esc(p.priceId)}" onclick="openPriceAdmin(this.dataset.priceId)">تعديل</button>
        </div>
      </div>`;
  }).join('');
}

  document.getElementById('mPriceDestination').value=p.destination||'';
  document.getElementById('mPriceService').value=p.serviceType||'توصيلة';
  document.getElementById('mPriceAmount').value=p.price||'';
  document.getElementById('mPriceWait').value=p.includedWaitMinutes==null?60:p.includedWaitMinutes;
  document.getElementById('mPriceExtraWait').value=p.extraWaitHourPrice==null?75:p.extraWaitHourPrice;
  document.getElementById('mPriceNotes').value=p.notes||'';
  document.getElementById('mPriceActive').value=p.active?'TRUE':'FALSE';
  document.getElementById('priceModal').classList.add('active');
}
