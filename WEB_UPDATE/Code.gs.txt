function doGet(e) {
  const view = String((e && e.parameter && e.parameter.view) || '').toLowerCase();
  const file = view === 'control' ? 'AdminV2' : (view === 'admin' ? 'Admin' : 'Index');
  const title = view === 'control'
    ? 'مشوارك | مركز التحكم'
    : (view === 'admin' ? 'مشوارك | لوحة الإدارة' : APP.NAME);

  return HtmlService.createTemplateFromFile(file)
    .evaluate()
    .setTitle(title)
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1, maximum-scale=1, user-scalable=no');
}

function normalizeArabicForPrice_(v) {
  return clean_(v)
    .toLowerCase()
    .replace(/[ًٌٍَُِّْـ]/g,'')
    .replace(/[أإآ]/g,'ا')
    .replace(/ى/g,'ي')
    .replace(/ة/g,'ه')
    .replace(/[^\u0600-\u06FF0-9a-zA-Z]+/g,' ')
    .replace(/\s+/g,' ')
    .trim();
}

function parseWaitingMinutes_(v) {
  const s=normalizeArabicForPrice_(v);
  if(!s) return 0;
  if(s.indexOf('اكثر من ساعتين')!==-1) return -1;
  if(s.indexOf('120')!==-1 || s.indexOf('ساعتان')!==-1 || s.indexOf('ساعتين')!==-1) return 120;
  if(s.indexOf('90')!==-1 || s.indexOf('ساعه ونصف')!==-1) return 90;
  if(s.indexOf('60')!==-1 || s==='ساعه') return 60;
  if(s.indexOf('30')!==-1 || s.indexOf('نصف ساعه')!==-1) return 30;
  if(s.indexOf('15')!==-1 || s.indexOf('ربع ساعه')!==-1) return 15;
  return 0;
}

function findBestPriceForDestination_(to) {
  const rows=getPublicPrices();
  const target=normalizeArabicForPrice_(to);
  if(!target) return null;

  // Never guess from similar words. We only price when the configured
  // destination appears explicitly in the customer's destination text,
  // or the two normalized strings are exactly equal.
  const candidates=[];

  rows.forEach(function(p){
    const d=normalizeArabicForPrice_(p.destination);
    if(!d) return;

    if(target===d){
      candidates.push({p:p,score:10000+d.length});
      return;
    }

    // Require the whole configured destination phrase to be present.
    // Padding with spaces prevents partial-word matches such as
    // قويسنا accidentally matching a different destination.
    const paddedTarget=' '+target+' ';
    const paddedDest=' '+d+' ';
    if(paddedTarget.indexOf(paddedDest)!==-1){
      candidates.push({p:p,score:5000+d.length});
    }
  });

  if(!candidates.length) return null;

  // Prefer the most specific/longest exact phrase when several are present.
  candidates.sort(function(a,b){return b.score-a.score;});
  return candidates[0].p;
}
function calculateEstimatedRidePrice_(to,tripType,waitingDetails,acPreference) {
  const match=findBestPriceForDestination_(to);
  if(!match){
    const acNote = String(acPreference||'').toUpperCase()==='WITH_AC'
      ? ' تشغيل التكييف قد يضيف تكلفة بسيطة، ويتم تأكيدها قبل تنفيذ المشوار.'
      : '';
    return {
      estimatedPrice:'',
      matchedDestination:'',
      note:'الوجهة غير موجودة حاليًا في قائمة الأسعار، وسيتم تأكيد السعر هاتفيًا.' + acNote
    };
  }

  let total=Number(match.price||0);
  const included=Number(match.includedWaitMinutes||0);
  const extraHour=Number(match.extraWaitHourPrice||75);
  const requested=parseWaitingMinutes_(waitingDetails);
  let waitingNote='';

  if(requested===-1){
    waitingNote=' مدة الانتظار الأكبر من ساعتين يتم تسعيرها عند التواصل.';
  }else if(requested>included){
    const extraMinutes=requested-included;
    total += Math.round((extraMinutes/60)*extraHour);
    waitingNote=' يشمل الحساب تكلفة الانتظار الإضافي المحدد.';
  }

  let acNote='';
  if(String(acPreference||'').toUpperCase()==='WITH_AC'){
    // No fixed AC surcharge is calculated automatically.
    acNote=' تشغيل التكييف قد يضيف تكلفة بسيطة، ويتم تأكيدها قبل تنفيذ المشوار.';
  }

  let note='سعر مبدئي بناءً على قائمة الأسعار، ويتم تأكيد السعر النهائي هاتفيًا.';
  if(tripType==='ROUND'){
    note+=' السعر الظاهر للذهاب؛ تكلفة العودة يتم تأكيدها عند التواصل.';
  }
  note+=waitingNote;
  note+=acNote;

  return {
    estimatedPrice:Math.round(total),
    matchedDestination:match.destination,
    note:note
  };
}



function roundMoneyV28_(value) {
  return Math.round(Number(value || 0) * 100) / 100;
}

function calculateRideFinanceV28_(amount) {
  const price = Math.max(0, roundMoneyV28_(amount));
  const rate = Number(FINANCE_RULES.COMMISSION_PERCENT || 10);
  const commission = roundMoneyV28_(price * rate / 100);
  const driverDue = roundMoneyV28_(price - commission);

  return {
    price: price,
    commissionRate: rate,
    commission: commission,
    driverDue: driverDue
  };
}

function normalizeRidePointV530_(latValue, lngValue) {
  const lat = Number(latValue);
  const lng = Number(lngValue);
  if (!isFinite(lat) || !isFinite(lng)) return null;
  if (lat < -90 || lat > 90 || lng < -180 || lng > 180) return null;
  return {
    lat: Math.round(lat * 1000000) / 1000000,
    lng: Math.round(lng * 1000000) / 1000000
  };
}

function buildRideMapUrlV530_(lat, lng) {
  return 'https://www.google.com/maps/search/?api=1&query=' +
    encodeURIComponent(String(lat) + ',' + String(lng));
}

function extractCoordsFromMapTextV530_(raw) {
  let text = String(raw || '').trim();
  if (!text) return null;
  try { text = decodeURIComponent(text.replace(/\+/g, ' ')); } catch (_) {}
  const patterns = [
    /(?:query|q|ll|destination)=\s*(-?\d{1,2}(?:\.\d+)?)\s*,\s*(-?\d{1,3}(?:\.\d+)?)/i,
    /@\s*(-?\d{1,2}(?:\.\d+)?)\s*,\s*(-?\d{1,3}(?:\.\d+)?)/,
    /geo:\s*(-?\d{1,2}(?:\.\d+)?)\s*,\s*(-?\d{1,3}(?:\.\d+)?)/i,
    /(?:^|[^\d.-])(-?\d{1,2}\.\d{4,})\s*,\s*(-?\d{1,3}\.\d{4,})(?:[^\d.]|$)/
  ];
  for (let i = 0; i < patterns.length; i++) {
    const m = text.match(patterns[i]);
    if (!m) continue;
    const p = normalizeRidePointV530_(m[1], m[2]);
    if (p) return p;
  }
  return null;
}

function isAllowedMapHostV530_(url) {
  const m = String(url || '').match(/^https:\/\/([^\/?#]+)/i);
  if (!m) return false;
  const host = String(m[1] || '').toLowerCase().replace(/:\d+$/, '');
  return [
    'maps.app.goo.gl',
    'maps.google.com',
    'www.google.com',
    'google.com',
    'goo.gl'
  ].indexOf(host) >= 0;
}

/**
 * Public helper used only when the customer pastes a Google Maps short link.
 * Fetching is restricted to Google Maps hosts to avoid turning the script into
 * a generic URL fetch/proxy endpoint.
 */
function resolveMashwarakMapLinkV530(raw) {
  const direct = extractCoordsFromMapTextV530_(raw);
  if (direct) return {success:true, lat:direct.lat, lng:direct.lng, resolved:false};

  let url = clean_(raw);
  const urlMatch = url.match(/https:\/\/[^\s]+/i);
  if (urlMatch) url = urlMatch[0];
  if (!isAllowedMapHostV530_(url)) {
    return {success:false, error:'Unsupported map link'};
  }
  // goo.gl is accepted only for the historical /maps short-link path.
  if (/^https:\/\/goo\.gl\//i.test(url) && !/^https:\/\/goo\.gl\/maps(?:\/|$)/i.test(url)) {
    return {success:false, error:'Unsupported short link'};
  }

  let current = url;
  for (let i = 0; i < 6; i++) {
    const point = extractCoordsFromMapTextV530_(current);
    if (point) return {success:true, lat:point.lat, lng:point.lng, resolved:i > 0};
    if (!isAllowedMapHostV530_(current)) break;

    let response;
    try {
      response = UrlFetchApp.fetch(current, {
        method:'get',
        followRedirects:false,
        muteHttpExceptions:true,
        headers:{'User-Agent':'Mashwarak/1.0 LocationResolver'}
      });
    } catch (e) {
      return {success:false, error:'Map link fetch failed'};
    }

    const headers = response.getHeaders ? response.getHeaders() : {};
    let location = '';
    Object.keys(headers || {}).some(function(k){
      if (String(k).toLowerCase() === 'location') {
        location = String(headers[k] || '');
        return true;
      }
      return false;
    });
    if (!location) {
      const bodyPoint = extractCoordsFromMapTextV530_(response.getContentText ? response.getContentText() : '');
      if (bodyPoint) return {success:true, lat:bodyPoint.lat, lng:bodyPoint.lng, resolved:true};
      break;
    }
    if (!/^https:\/\//i.test(location)) break;
    if (!isAllowedMapHostV530_(location)) break;
    current = location;
  }

  const finalPoint = extractCoordsFromMapTextV530_(current);
  return finalPoint
    ? {success:true, lat:finalPoint.lat, lng:finalPoint.lng, resolved:true}
    : {success:false, error:'Coordinates not found'};
}

function submitRideRequest(payload) {
  ensureReady_();
  payload = payload || {};

  const name = clean_(payload.customerName);
  const mobile = normalizePhone_(payload.mobile);
  const tripType = payload.tripType === 'ROUND' ? 'ROUND' : 'ONE_WAY';
  const from = clean_(payload.from);
  const to = clean_(payload.to);
  const date = clean_(payload.date);
  const time = clean_(payload.time);
  const passengers = Math.max(1, Math.min(20, Number(payload.passengers || 1)));
  const notes = clean_(payload.notes);
  const returnDate = clean_(payload.returnDate);
  const returnTime = clean_(payload.returnTime);
  const waiting = String(payload.waiting || 'NO').toUpperCase() === 'YES' ? 'YES' : 'NO';
  const waitingDetails = clean_(payload.waitingDetails);
  const termsAccepted = payload.termsAccepted === true || String(payload.termsAccepted).toUpperCase() === 'TRUE';
  const extras = Array.isArray(payload.extras) ? payload.extras.map(clean_).filter(Boolean) : [];
  const captainAgePreference = String(payload.captainAgePreference || 'NO').toUpperCase() === 'YES' ? 'YES' : 'NO';
  const captainAgeMin = clean_(payload.captainAgeMin);
  const captainAgeMax = clean_(payload.captainAgeMax);
  const acPreferenceRaw = String(payload.acPreference || '').toUpperCase();
  const acPreference = acPreferenceRaw === 'WITH_AC' ? 'WITH_AC' : 'WITHOUT_AC';
  const requestId = clean_(payload.requestId).slice(0, 120);
  const fromPoint = normalizeRidePointV530_(payload.fromLat, payload.fromLng);
  const toPoint = normalizeRidePointV530_(payload.toLat, payload.toLng);
  const fromMapLabel = clean_(payload.fromMapLabel).slice(0, 250);
  const toMapLabel = clean_(payload.toMapLabel).slice(0, 250);
  const fromMapUrl = fromPoint ? buildRideMapUrlV530_(fromPoint.lat, fromPoint.lng) : '';
  const toMapUrl = toPoint ? buildRideMapUrlV530_(toPoint.lat, toPoint.lng) : '';

  if (!name) throw new Error('من فضلك اكتب الاسم.');
  if (mobile.length < 8) throw new Error('من فضلك اكتب رقم موبايل صحيح.');
  if (!from) throw new Error('حدد مكان الانطلاق.');
  if (!to) throw new Error('حدد الوجهة.');
  if (!date) throw new Error('حدد تاريخ المشوار.');
  if (!time) throw new Error('حدد وقت المشوار.');
  if (tripType === 'ROUND' && !returnDate) throw new Error('حدد تاريخ العودة.');
  if (tripType === 'ROUND' && !returnTime) throw new Error('حدد وقت العودة.');
  if (waiting === 'YES' && !waitingDetails) throw new Error('اكتب تفاصيل الانتظار.');
  if (!termsAccepted) throw new Error('يجب الموافقة على الشروط والأحكام.');
  if (captainAgePreference === 'YES') {
    if (!captainAgeMin || !captainAgeMax) throw new Error('اختر نطاق سن الكابتن المفضل.');
    if (Number(captainAgeMin) > Number(captainAgeMax)) throw new Error('نطاق سن الكابتن غير صحيح.');
  }

  const now = new Date();
  // V5.3.0 Smart Location phase 1:
  // destination-list automatic pricing is intentionally disabled. The next
  // pricing phase will use actual driving-route distance x configured EGP/km.
  const estimate = {
    estimatedPrice:'',
    matchedDestination:'',
    note:'التسعير التلقائي غير مفعل حاليًا.'
  };
  const finance = {price:0,driverDue:0,commission:0,commissionRate:Number(FINANCE_RULES.COMMISSION_PERCENT || 10)};
  const sh = getDb_().getSheetByName(APP.SHEETS.RIDES);
  if (!sh) throw new Error('جدول طلبات المشاوير غير موجود.');

  ensureNamedHeaders_(sh, [
    'Return Date','Return Time','Waiting','Waiting Details','Terms Accepted',
    'Extras','Captain Age Preference','Captain Age Min','Captain Age Max',
    'Customer AC Preference','From Latitude','From Longitude','From Map URL','From Map Label',
    'To Latitude','To Longitude','To Map URL','To Map Label',
    'Driver Cost','Mashwarak Commission','Commission Rate','Workflow Stage','Contacted At','Confirmed At','Assigned At','Started At','Arrived At','Completed At','Cancelled At','Cancel Reason','Client Request ID'
  ]);

  // Generate the candidate ID before the write lock. nextId_ uses the script
  // lock internally, so keeping it outside avoids nested-lock deadlocks.
  const candidateId = nextId_('R', APP.SHEETS.RIDES);
  const writeLock = LockService.getScriptLock();
  writeLock.waitLock(15000);
  let id = candidateId;
  try {
    if (requestId) {
      const existing = findRideByClientRequestIdV242_(sh, requestId);
      if (existing) {
        return {
          success:true,
          rideId:existing.rideId,
          estimatedPrice:existing.estimatedPrice || estimate.estimatedPrice || '',
          matchedDestination:estimate.matchedDestination || '',
          estimateNote:estimate.note || '',
          acSurcharge:0,
          duplicate:true,
          message:'تم استلام الطلب بالفعل.'
        };
      }
    }

    appendRideNamedRowV26_(sh, {
      'Ride ID':id,
      'Created At':now,
      'Customer Name':name,
      'Mobile':mobile,
      'Trip Type':tripType,
      'From':from,
      'To':to,
      'Date':date,
      'Time':time,
      'Passengers':passengers,
      'Notes':notes,
      'Estimated Price':estimate.estimatedPrice || '',
      'Final Price':'',
      'Assigned Driver ID':'',
      'Assigned Driver Name':'',
      'Status':'NEW',
      'Admin Notes':'',
      'Updated At':now,
      'Return Date':returnDate,
      'Return Time':returnTime,
      'Waiting':waiting,
      'Waiting Details':waitingDetails,
      'Terms Accepted':termsAccepted,
      'Extras':extras.join(' | '),
      'Captain Age Preference':captainAgePreference,
      'Captain Age Min':captainAgeMin,
      'Captain Age Max':captainAgeMax,
      'Customer AC Preference':acPreference,
      'From Latitude':fromPoint ? fromPoint.lat : '',
      'From Longitude':fromPoint ? fromPoint.lng : '',
      'From Map URL':fromMapUrl,
      'From Map Label':fromMapLabel,
      'To Latitude':toPoint ? toPoint.lat : '',
      'To Longitude':toPoint ? toPoint.lng : '',
      'To Map URL':toMapUrl,
      'To Map Label':toMapLabel,
      'Driver Cost':'',
      'Mashwarak Commission':'',
      'Commission Rate':finance.commissionRate,
      'Workflow Stage':'NEW',
      'Client Request ID':requestId
    });
  } finally {
    writeLock.releaseLock();
  }

  // A ride is considered accepted as soon as the durable RideRequests row is
  // written. Logging, customer aggregation and monitor FCM must never block
  // the client. V5.2.4 restores immediate monitor notifications: try direct
  // NEW_RIDE push first, and only fall back to the retry queue on failure.
  try {
    log_('CREATE_RIDE', 'RIDE', id, JSON.stringify({from:from,to:to,mobile:mobile,acPreference:acPreference,requestId:requestId}));
  } catch (logErr) {}
  try { invalidateControlSnapshotV23_(); } catch (cacheErr) {}

  try {
    upsertCustomerFromRide_(id, name, mobile, now);
  } catch (customerErr) {
    try { log_('CUSTOMER_UPSERT_ERROR','RIDE',id,String(customerErr)); } catch (_) {}
  }

  let instantRidePushSent = false;
  try {
    const directPush = sendMashwarakAdminPush_(
      'طلب مشوار جديد',
      String(name || '') + ' | ' + String(from || '') + ' ← ' + String(to || ''),
      'NEW_RIDE',
      id
    );
    if (directPush && directPush.success) {
      instantRidePushSent = true;
      try { log_('ADMIN_PUSH_SENT','RIDE',id,'instant monitor topic mashwarak_admin_all'); } catch (_) {}
    } else {
      throw new Error((directPush && directPush.error) || 'FCM admin push not sent');
    }
  } catch (pushErr) {
    try { log_('ADMIN_PUSH_FALLBACK_QUEUE','RIDE',id,String(pushErr)); } catch (_) {}
    try {
      queueRidePostSubmitV242_({
        rideId:id,
        name:name,
        mobile:mobile,
        from:from,
        to:to,
        when:now.getTime(),
        retryCount:0,
        lastError:String(pushErr && pushErr.message ? pushErr.message : pushErr)
      });
    } catch (queueErr) {
      try { log_('POST_SUBMIT_QUEUE_ERROR','RIDE',id,String(queueErr)); } catch (_) {}
    }
  }

  return {
    success: true,
    rideId: id,
    estimatedPrice: estimate.estimatedPrice || '',
    matchedDestination: estimate.matchedDestination || '',
    estimateNote: estimate.note || '',
    acSurcharge:0,
    duplicate:false,
    message: 'تم استلام طلبك بنجاح وسيتم التواصل معك هاتفيًا لتأكيد التفاصيل.'
  };
}

function findRideByClientRequestIdV242_(sh, requestId) {
  requestId = clean_(requestId);
  if (!sh || !requestId || sh.getLastRow() < 2) return null;

  const lastCol = sh.getLastColumn();
  const headers = sh.getRange(1,1,1,lastCol).getDisplayValues()[0];
  const requestCol = headers.indexOf('Client Request ID') + 1;
  if (!requestCol) return null;

  const match = sh.getRange(2,requestCol,sh.getLastRow()-1,1)
    .createTextFinder(requestId)
    .matchEntireCell(true)
    .findNext();
  if (!match) return null;

  const row = sh.getRange(match.getRow(),1,1,lastCol).getValues()[0];
  const get = function(header) {
    const idx = headers.indexOf(header);
    return idx >= 0 ? row[idx] : '';
  };
  return {
    rideId:String(get('Ride ID') || ''),
    estimatedPrice:get('Estimated Price') || ''
  };
}

function queueRidePostSubmitV242_(item) {
  const props = PropertiesService.getScriptProperties();
  const rideId = clean_(item && item.rideId);
  if (!rideId) return;

  const queued = Object.assign({}, item || {}, {
    rideId:rideId,
    retryCount:Number((item && item.retryCount) || 0),
    queuedAt:Date.now()
  });
  props.setProperty('MASHWARAK_POST_RIDE_' + rideId, JSON.stringify(queued));

  // V5.2.3: keep ride acceptance independent from FCM, but make sure there is
  // a persistent one-minute worker instead of relying on a fragile one-shot
  // trigger for every request. If Apps Script needs a new authorization scope,
  // this attempt may fail silently here; run setupMashwarakMonitorNotificationsV523()
  // once from the editor to authorize/install it permanently.
  try { ensureRidePostSubmitWorkerV523_(); } catch (triggerErr) {
    try { log_('POST_SUBMIT_TRIGGER_ERROR','RIDE',rideId,String(triggerErr)); } catch (_) {}
  }
}

function ensureRidePostSubmitWorkerV523_() {
  const handler = 'processRidePostSubmitQueueV242_';
  const triggers = ScriptApp.getProjectTriggers();
  const exists = triggers.some(function(t){
    return t.getHandlerFunction && t.getHandlerFunction() === handler;
  });
  if (exists) return true;

  ScriptApp.newTrigger(handler)
    .timeBased()
    .everyMinutes(1)
    .create();
  return true;
}

// Kept for compatibility with V5.2.2 calls/backups.
function scheduleRidePostSubmitV242_() {
  return ensureRidePostSubmitWorkerV523_();
}

function processRidePostSubmitQueueV242_() {
  const props = PropertiesService.getScriptProperties();
  props.deleteProperty('MASHWARAK_POST_RIDE_TRIGGER_AT'); // stale V5.2.2 marker

  const all = props.getProperties();
  const keys = Object.keys(all)
    .filter(function(k){ return k.indexOf('MASHWARAK_POST_RIDE_') === 0; })
    .slice(0,30);

  let sent = 0;
  let retried = 0;
  let dropped = 0;

  keys.forEach(function(key){
    let item = null;
    try { item = JSON.parse(all[key] || '{}'); } catch (_) {}
    if (!item || !item.rideId) {
      props.deleteProperty(key);
      dropped++;
      return;
    }

    const when = item.when ? new Date(Number(item.when)) : new Date();
    try {
      upsertCustomerFromRide_(item.rideId, item.name, item.mobile, when);
    } catch (customerErr) {
      try { log_('CUSTOMER_UPSERT_ERROR','RIDE',item.rideId,String(customerErr)); } catch (_) {}
      // Customer aggregation is secondary and must not prevent the alert.
    }

    try {
      const pushResult = sendMashwarakAdminPush_(
        'طلب مشوار جديد',
        String(item.name || '') + ' | ' + String(item.from || '') + ' ← ' + String(item.to || ''),
        'NEW_RIDE',
        item.rideId
      );
      if (pushResult && pushResult.success) {
        props.deleteProperty(key);
        sent++;
        try { log_('ADMIN_PUSH_SENT','RIDE',item.rideId,'monitor topic mashwarak_admin_all'); } catch (_) {}
        return;
      }
      throw new Error((pushResult && pushResult.error) || 'FCM admin push not sent');
    } catch (pushErr) {
      item.retryCount = Number(item.retryCount || 0) + 1;
      item.lastError = String(pushErr && pushErr.message ? pushErr.message : pushErr).slice(0,500);
      item.lastAttemptAt = Date.now();

      // Keep the queue item so a temporary Firebase/network error does not lose
      // the monitor notification. Cap retries to avoid an eternal broken item.
      if (item.retryCount >= 20) {
        props.deleteProperty(key);
        dropped++;
        try { log_('ADMIN_PUSH_DROPPED','RIDE',item.rideId,item.lastError); } catch (_) {}
      } else {
        props.setProperty(key, JSON.stringify(item));
        retried++;
        try { log_('ADMIN_PUSH_RETRY','RIDE',item.rideId,'try=' + item.retryCount + ' | ' + item.lastError); } catch (_) {}
      }
    }
  });

  return {success:true,processed:keys.length,sent:sent,retried:retried,dropped:dropped};
}

/**
 * Run ONCE manually from Apps Script after deploying V5.2.3.
 * It authorizes/installs the persistent worker, flushes any rides that were
 * queued by V5.2.2, and sends a direct test notification to the Monitor app.
 */
function setupMashwarakMonitorNotificationsV523() {
  ensureReady_();
  const props = PropertiesService.getScriptProperties();
  const projectId = props.getProperty('MASHWARAK_FCM_PROJECT_ID');
  const clientEmail = props.getProperty('MASHWARAK_FCM_CLIENT_EMAIL');
  const privateKey = props.getProperty('MASHWARAK_FCM_PRIVATE_KEY');
  if (!projectId || !clientEmail || !privateKey) {
    throw new Error('بيانات Firebase Service Account غير مكتملة في Script Properties.');
  }

  // Remove obsolete duplicate workers, then create exactly one repeating worker.
  const handler = 'processRidePostSubmitQueueV242_';
  ScriptApp.getProjectTriggers().forEach(function(t){
    if (t.getHandlerFunction && t.getHandlerFunction() === handler) {
      ScriptApp.deleteTrigger(t);
    }
  });
  ScriptApp.newTrigger(handler).timeBased().everyMinutes(1).create();
  props.deleteProperty('MASHWARAK_POST_RIDE_TRIGGER_AT');

  const flush = processRidePostSubmitQueueV242_();
  const test = sendMashwarakAdminPush_(
    'اختبار تطبيق إشعارات مشوارك',
    'تم إصلاح مسار إشعارات المتابعة بنجاح.',

  );

  return {
    success:true,
    version:APP.VERSION,
    topic:'mashwarak_admin_all',
    workerInstalled:true,
    queueFlush:flush,
    testPush:test
  };
}

  ensureReady_();

  const name = clean_(payload.fullName);
  const mobile = normalizePhone_(payload.mobile);
  const city = clean_(payload.cityArea);
  const carType = clean_(payload.carType);
  const brandModel = clean_(payload.brandModel);
  const year = clean_(payload.year);
  const color = clean_(payload.color);
  const seats = Math.max(1, Math.min(30, Number(payload.seats || 4)));
  const acWorking = String(payload.acWorking || 'YES').toUpperCase() === 'NO' ? 'NO' : 'YES';
  const availabilityType = String(payload.availabilityType || 'FULL_TIME').toUpperCase() === 'LIMITED' ? 'LIMITED' : 'FULL_TIME';

    success:true,
    id:id,
    pushSent:pushSent,
    pushError:pushError
  };
}

/**
 * Run once from Apps Script editor after creating the Firebase service account.
 *
 * Example:
 * configureMashwarakFirebase(
 *   'your-project-id',
 *   'firebase-adminsdk-xxxx@your-project-id.iam.gserviceaccount.com',
 *   '-----BEGIN PRIVATE KEY-----\n...\n-----END PRIVATE KEY-----\n'
 * );
 */
function configureMashwarakFirebase(projectId, clientEmail, privateKey) {
  projectId = clean_(projectId);
  clientEmail = clean_(clientEmail);
  privateKey = String(privateKey || '').trim();

  if (!projectId) throw new Error('Firebase Project ID غير موجود.');
  if (!clientEmail || clientEmail.indexOf('@') === -1) {
    throw new Error('Firebase Client Email غير صحيح.');
  }
  if (privateKey.indexOf('BEGIN PRIVATE KEY') === -1) {
    throw new Error('Firebase Private Key غير صحيح.');
  }

  const props = PropertiesService.getScriptProperties();
  props.setProperty('MASHWARAK_FCM_PROJECT_ID', projectId);
  props.setProperty('MASHWARAK_FCM_CLIENT_EMAIL', clientEmail);
  props.setProperty('MASHWARAK_FCM_PRIVATE_KEY', privateKey.replace(/\\n/g, '\n'));

  CacheService.getScriptCache().remove('MASHWARAK_FCM_ACCESS_TOKEN');
  return {success:true};
}

function clearMashwarakFirebaseConfig() {
  const props = PropertiesService.getScriptProperties();
  [
    'MASHWARAK_FCM_PROJECT_ID',
    'MASHWARAK_FCM_CLIENT_EMAIL',
    'MASHWARAK_FCM_PRIVATE_KEY'
  ].forEach(function(k){ props.deleteProperty(k); });

  CacheService.getScriptCache().remove('MASHWARAK_FCM_ACCESS_TOKEN');
  return {success:true};
}

function getMashwarakFirebaseStatus() {
  const props = PropertiesService.getScriptProperties();
  return {
    configured: !!(
      props.getProperty('MASHWARAK_FCM_PROJECT_ID') &&
      props.getProperty('MASHWARAK_FCM_CLIENT_EMAIL') &&
      props.getProperty('MASHWARAK_FCM_PRIVATE_KEY')
    ),
    projectId: props.getProperty('MASHWARAK_FCM_PROJECT_ID') || '',
    topic: 'mashwarak_all'
  };
}

function sendMashwarakPush_(title, message, notificationId) {
  const props = PropertiesService.getScriptProperties();
  const projectId = props.getProperty('MASHWARAK_FCM_PROJECT_ID');

  if (!projectId) {
    return {
      success:false,
      skipped:true,
      error:'Firebase غير مهيأ بعد.'
    };
  }

  const accessToken = getMashwarakFcmAccessToken_();
  const url = 'https://fcm.googleapis.com/v1/projects/' +
    encodeURIComponent(projectId) + '/messages:send';

  const payload = {
    message: {
      topic: 'mashwarak_all',
      data: {
        title: String(title || 'مشوارك'),
        message: String(message || ''),
        notificationId: String(notificationId || '')
      },
      android: {
        priority: 'HIGH'
      }
    }
  };

  const res = UrlFetchApp.fetch(url, {
    method:'post',
    contentType:'application/json',
    payload:JSON.stringify(payload),
    headers:{
      Authorization:'Bearer ' + accessToken
    },
    muteHttpExceptions:true
  });

  const status = res.getResponseCode();
  const body = res.getContentText();

  if (status < 200 || status >= 300) {
    throw new Error('تعذر إرسال Push. FCM HTTP ' + status + ': ' + body.slice(0,500));
  }

  return {success:true,response:body};
}

function getMashwarakFcmAccessToken_() {
  const cache = CacheService.getScriptCache();
  const cached = cache.get('MASHWARAK_FCM_ACCESS_TOKEN');
  if (cached) return cached;

  const props = PropertiesService.getScriptProperties();
  const clientEmail = props.getProperty('MASHWARAK_FCM_CLIENT_EMAIL');
  const privateKey = props.getProperty('MASHWARAK_FCM_PRIVATE_KEY');

  if (!clientEmail || !privateKey) {
    throw new Error('بيانات Firebase Service Account غير مكتملة.');
  }

  const now = Math.floor(Date.now() / 1000);
  const header = {
    alg:'RS256',
    typ:'JWT'
  };
  const claim = {
    iss:clientEmail,
    scope:'https://www.googleapis.com/auth/firebase.messaging',
    aud:'https://oauth2.googleapis.com/token',
    iat:now,
    exp:now + 3600
  };

  const encodedHeader = base64UrlJson_(header);
  const encodedClaim = base64UrlJson_(claim);
  const unsignedJwt = encodedHeader + '.' + encodedClaim;

  const signature = Utilities.computeRsaSha256Signature(
    unsignedJwt,
    privateKey
  );

  const jwt = unsignedJwt + '.' + Utilities.base64EncodeWebSafe(signature)
    .replace(/=+$/,'');

  const tokenRes = UrlFetchApp.fetch('https://oauth2.googleapis.com/token', {
    method:'post',
    contentType:'application/x-www-form-urlencoded',
    payload:{
      grant_type:'urn:ietf:params:oauth:grant-type:jwt-bearer',
      assertion:jwt
    },
    muteHttpExceptions:true
  });

  const status = tokenRes.getResponseCode();
  const body = tokenRes.getContentText();

  if (status < 200 || status >= 300) {
    throw new Error('تعذر الحصول على Firebase access token. HTTP ' + status + ': ' + body.slice(0,500));
  }

  const parsed = JSON.parse(body);
  const token = parsed.access_token;
  if (!token) throw new Error('Firebase access token غير موجود في الاستجابة.');

  cache.put('MASHWARAK_FCM_ACCESS_TOKEN', token, 3300);
  return token;
}

function base64UrlJson_(obj) {
  return Utilities.base64EncodeWebSafe(
    Utilities.newBlob(JSON.stringify(obj)).getBytes()
  ).replace(/=+$/,'');
}

function ensureNamedHeaders_(sh, names) {
  if (!sh) return;
  const lastCol = Math.max(1, sh.getLastColumn());
  const current = sh.getRange(1,1,1,lastCol).getDisplayValues()[0];
  names.forEach(function(name) {
    if (current.indexOf(name) === -1) {
      sh.getRange(1, sh.getLastColumn() + 1).setValue(name);
      current.push(name);
    }
  });
  sh.setFrozenRows(1);
}

function appendRideNamedRowV26_(sh, valuesByHeader) {
  const lastCol = sh.getLastColumn();
  const headers = sh.getRange(1,1,1,lastCol).getDisplayValues()[0];
  const row = new Array(lastCol).fill('');

  headers.forEach(function(header, i){
    if (Object.prototype.hasOwnProperty.call(valuesByHeader, header)) {
      row[i] = valuesByHeader[header];
    }
  });

  sh.appendRow(row);
}

    if (Object.prototype.hasOwnProperty.call(valuesByHeader, header)) {
      row[i] = valuesByHeader[header];
    }
  });
  sh.appendRow(row);
}

    if (!item || !item.data) throw new Error('مستند ناقص: ' + key);

    const mime = clean_(item.mimeType) || 'image/jpeg';
    if (['image/jpeg','image/png','image/webp'].indexOf(mime) === -1) {
      throw new Error('نوع ملف غير مدعوم. ارفع صورة JPG أو PNG أو WEBP.');
    }

    const bytes = Utilities.base64Decode(String(item.data));
    if (bytes.length > 3 * 1024 * 1024) {
      throw new Error('إحدى الصور أكبر من الحد المسموح بعد الضغط.');
    }

    const ext = mime === 'image/png' ? '.png' : mime === 'image/webp' ? '.webp' : '.jpg';
    const blob = Utilities.newBlob(bytes, mime, labels[key] + ext);
    const f = folder.createFile(blob);
    out[key] = f.getUrl();
  });

  return out;
}

function getAdminDashboard_(token) {
  assertAdmin_(token);

  const rides = getDb_().getSheetByName(APP.SHEETS.RIDES).getDataRange().getValues();
  const drivers = getDb_().getSheetByName(APP.SHEETS.DRIVERS).getDataRange().getValues();

  const stats = {
    totalRides: 0,
    newRides: 0,
    confirmed: 0,
    assigned: 0,
    completed: 0,
    cancelled: 0,
    drivers: 0,
    driversUnderReview: 0,
    approvedDrivers: 0
  };

  for (let i = 1; i < rides.length; i++) {
    if (!rides[i][0]) continue;
    stats.totalRides++;
    const s = String(rides[i][15] || 'NEW').toUpperCase();
    if (s === 'NEW') stats.newRides++;
    if (s === 'CONFIRMED') stats.confirmed++;
    if (s === 'ASSIGNED') stats.assigned++;
    if (s === 'COMPLETED') stats.completed++;
    if (s === 'CANCELLED') stats.cancelled++;
  }

  for (let i = 1; i < drivers.length; i++) {
    if (!drivers[i][0]) continue;
    stats.drivers++;
    const s = String(drivers[i][14] || 'UNDER_REVIEW').toUpperCase();
    if (s === 'UNDER_REVIEW') stats.driversUnderReview++;
    if (s === 'APPROVED') stats.approvedDrivers++;
  }

  return {
    stats: stats,
    recentRides: getAdminRides(token, 'ALL').slice(0, 8),
    recentDrivers: getAdminDrivers(token, 'ALL').slice(0, 8)
  };
}

function nextId_(prefix, sheetName) {
  const props = PropertiesService.getScriptProperties();
  const key = 'SEQ_' + sheetName;
  const lock = LockService.getScriptLock();

  try {
    let n = Number(props.getProperty(key) || 0) + 1;
    props.setProperty(key, String(n));
    const y = Utilities.formatDate(new Date(), APP.TZ, 'yy');
    return prefix + y + '-' + String(n).padStart(4,'0');
  } finally {
    lock.releaseLock();
  }
}

function hash_(value) {
  const bytes = Utilities.computeDigest(
    Utilities.DigestAlgorithm.SHA_256,
    String(value || ''),
    Utilities.Charset.UTF_8

  return bytes.map(b => ('0' + (b & 255).toString(16)).slice(-2)).join('');
}

function controlLogin(username, password) {
  ensureControlV2Ready_();

  username = clean_(username).toLowerCase();
  password = String(password || '');
  const sh = getDb_().getSheetByName(APP.SHEETS.ADMIN_USERS);
  const values = sh.getDataRange().getValues();

  for (let i=1;i<values.length;i++) {
    const r = values[i];

    if (!(r[5] === true || String(r[5]).toUpperCase() === 'TRUE')) {
      throw new Error('الحساب غير نشط.');
    }

      Utilities.sleep(250);
      throw new Error('اسم المستخدم أو كلمة المرور غير صحيحة.');
    }

    const user = {
      userId:String(r[0]),
      name:String(r[1] || ''),
      username:String(r[2] || ''),
      role:String(r[4] || 'VIEWER').toUpperCase()
    };

    const token = Utilities.getUuid();
    CacheService.getScriptCache().put(
      'CONTROL_' + token,
      JSON.stringify(user),
      21600
    );
    sh.getRange(i+1,8).setValue(new Date());
    logControlAction_(user,'LOGIN','USER',user.userId,'Control Center');

    return {success:true,token:String(token),user:user};
  }

  Utilities.sleep(250);
  throw new Error('اسم المستخدم أو كلمة المرور غير صحيحة.');

    } catch(e) {}
  }

        from:String(r[5] || ''),
        to:String(r[6] || ''),

      };
    }
  );

  const drivers = getControlAlertTailV27_(
    ss.getSheetByName(APP.SHEETS.DRIVERS),
    20,
    function(r){
      return {
        driverId:String(r[0] || ''),
        createdAt:formatAlertValueV27_(r[1]),
        fullName:String(r[2] || ''),
        mobile:String(r[3] || ''),
        cityArea:String(r[4] || ''),
        adminStatus:String(r[14] || 'UNDER_REVIEW').toUpperCase()
      };
    }
  );

    cached:false,
    dataEpoch:dataEpochV53_(),
    rides:rides,
    drivers:drivers,
    checkedAt:Utilities.formatDate(new Date(), APP.TZ, 'yyyy-MM-dd HH:mm:ss')
  };

  try {
    cache.put(cacheKey, JSON.stringify(out), 6);
  } catch(e) {}

  return out;
}

function getControlAlertTailV27_(sh, limit, mapper) {
  if (!sh) return [];

  const lastRow = sh.getLastRow();
  if (lastRow < 2) return [];

  const lastCol = sh.getLastColumn();
  const count = Math.min(Math.max(1, Number(limit || 20)), lastRow - 1);
  const startRow = Math.max(2, lastRow - count + 1);

  const rows = sh.getRange(startRow, 1, count, lastCol).getValues();
  const out = [];

  for (let i = rows.length - 1; i >= 0; i--) {
    const item = mapper(rows[i]);

    if (
      (item.rideId && String(item.rideId).trim()) ||
      (item.driverId && String(item.driverId).trim())
    ) {
      out.push(item);
    }
  }

  return out;
}

function formatAlertValueV27_(value) {
  if (value instanceof Date) {
    return Utilities.formatDate(value, APP.TZ, 'yyyy-MM-dd HH:mm:ss');
  }
  return String(value || '');
}

function getControlDashboard(token) {
  getControlUser_(token);
  ensureControlV2Ready_();

  const rides = getAdminRidesLegacyForV2_();
  const drivers = getAdminDriversLegacyForV2_();
  const tx = getTransactionsRawV2_();

  const today = Utilities.formatDate(new Date(), APP.TZ, 'yyyy-MM-dd');
  const commissionAccounts = calculateAllDriverCommissionAccountsV51_(rides,tx);

  let commissionCollectedToday = 0;
  let expensesToday = 0;
  let legacyDriverPaidToday = 0;

  tx.forEach(function(t){
    if (t.status !== 'CONFIRMED' || t.createdDate !== today) return;

    if (
      t.type === 'DRIVER_COMMISSION_PAYMENT' &&
      t.direction === 'IN'
    ) {
      commissionCollectedToday += Number(t.amount || 0);
    } else if (
      t.type === 'EXPENSE' &&
      t.direction === 'OUT'
    ) {
      expensesToday += Number(t.amount || 0);
    } else if (
      t.type === 'DRIVER_PAYMENT' &&
      t.direction === 'OUT'
    ) {
      legacyDriverPaidToday += Number(t.amount || 0);
    }
  });

  const counts = {
    NEW:0,CONTACTED:0,CONFIRMED:0,ASSIGNED:0,
    IN_PROGRESS:0,COMPLETED:0,CANCELLED:0
  };

  let completedCommissionTotal = 0;

  rides.forEach(function(r){
    if (Object.prototype.hasOwnProperty.call(counts,r.status)) counts[r.status]++;

    if (r.status === 'COMPLETED') {
      completedCommissionTotal += Number(r.mashwarakCommission || 0);
    }
  });

  let driverCommissionOutstanding = 0;
  Object.keys(commissionAccounts).forEach(function(driverId){
    driverCommissionOutstanding += Number(
      commissionAccounts[driverId].commissionOutstanding || 0
    );
  });

  let driversUnderReview = 0;
  let driversApproved = 0;

  drivers.forEach(function(d){
    if (d.adminStatus === 'UNDER_REVIEW') driversUnderReview++;
    if (d.adminStatus === 'APPROVED') driversApproved++;
  });

  return {
    ridesTotal:rides.length,
    newRides:counts.NEW,
    contactedRides:counts.CONTACTED,
    confirmedRides:counts.CONFIRMED,
    assignedRides:counts.ASSIGNED,
    inProgressRides:counts.IN_PROGRESS,
    completedRides:counts.COMPLETED,
    cancelledRides:counts.CANCELLED,

    driversTotal:drivers.length,
    driversUnderReview:driversUnderReview,
    driversApproved:driversApproved,

    revenueToday:roundMoneyV28_(commissionCollectedToday),
    commissionCollectedToday:roundMoneyV28_(commissionCollectedToday),
    completedCommissionTotal:roundMoneyV28_(completedCommissionTotal),
    driverCommissionOutstanding:roundMoneyV28_(driverCommissionOutstanding),

    expensesToday:roundMoneyV28_(expensesToday),
    driverPaidToday:roundMoneyV28_(legacyDriverPaidToday),
    netCashToday:roundMoneyV28_(
      commissionCollectedToday-expensesToday-legacyDriverPaidToday
    ),

    customerOutstanding:0
  };
}

function getControlRides(token, filters) {
  getControlUser_(token);
  filters = filters || {};

  let rows = getAdminRidesLegacyForV2_();
  const status = clean_(filters.status).toUpperCase();
  const q = normalizeSearchV2_(filters.q);

  if (status && status !== 'ALL') {
    rows = rows.filter(function(r){ return r.status === status; });
  }

  if (q) {
    rows = rows.filter(function(r){
      return normalizeSearchV2_([
        r.rideId,r.customerName,r.mobile,r.from,r.to,
        r.assignedDriverName,r.status
      ].join(' ')).indexOf(q) !== -1;
    });
  }

  return rows.map(function(r){
    r.customerPaymentMode = 'DIRECT_TO_DRIVER';
    r.customerPaid = 0;
    r.customerBalance = 0;
    r.driverPaid = 0;
    r.driverBalance = 0;
    r.captainNet = Number(r.driverCost || 0);
    r.rideCommissionDue = r.status === 'COMPLETED'
      ? Number(r.mashwarakCommission || 0)
      : 0;
    r.paymentStatus = 'DIRECT_TO_DRIVER';

    return r;
  });
}

function getControlDrivers(token, filters) {
  getControlUser_(token);
  filters = filters || {};

  let rows = getAdminDriversLegacyForV2_();
  const status = clean_(filters.status).toUpperCase();
  const q = normalizeSearchV2_(filters.q);

  if (status && status !== 'ALL') {
    rows = rows.filter(function(r){ return r.adminStatus === status; });
  }

  if (q) {
    rows = rows.filter(function(r){
      return normalizeSearchV2_([
        r.driverId,r.fullName,r.mobile,r.cityArea,
        r.brandModel,r.brand,r.model,r.adminStatus
      ].join(' ')).indexOf(q) !== -1;
    });
  }

  const rides = getAdminRidesLegacyForV2_();
  const tx = getTransactionsRawV2_();
  const accounts = calculateAllDriverCommissionAccountsV51_(rides,tx);

  const tripCounts = Object.create(null);
  const completedCounts = Object.create(null);

  rides.forEach(function(r){
    if (!r.assignedDriverId) return;

    tripCounts[r.assignedDriverId] =
      (tripCounts[r.assignedDriverId] || 0) + 1;

    if (r.status === 'COMPLETED') {
      completedCounts[r.assignedDriverId] =
        (completedCounts[r.assignedDriverId] || 0) + 1;
    }
  });

  return rows.map(function(d){
    const acc = accounts[d.driverId] || {
      commissionDue:0,
      commissionPaid:0,
      commissionOutstanding:0
    };

    d.completedTrips = Number(completedCounts[d.driverId] || 0);
    d.totalTrips = Number(tripCounts[d.driverId] || 0);
    d.commissionDue = Number(acc.commissionDue || 0);
    d.commissionPaid = Number(acc.commissionPaid || 0);
    d.commissionOutstanding = Number(acc.commissionOutstanding || 0);
    d.paidToDriver = 0;

    return d;
  });
}


function customerPhoneKeyV29_(value) {
  let d = String(value || '').replace(/\D/g,'');

  if (d.indexOf('0020') === 0) d = d.slice(4);
  if (d.indexOf('20') === 0 && d.length === 12) d = d.slice(2);
  if (d.indexOf('0') === 0 && d.length === 11) d = d.slice(1);

  return d;
}

function customerDisplayPhoneV29_(value) {
  const raw = String(value || '').trim();
  const key = customerPhoneKeyV29_(raw);

  if (/^1[0125]\d{8}$/.test(key)) {
    return '0' + key;
  }

  return raw;
}

    if (!stats[key]) {
      stats[key] = {
        rideIds:Object.create(null),
        total:0,
        completed:0,
        cancelled:0,
        lastRideAt:'',
        totalValue:0,
        paid:0,
        outstanding:0
      };
    }

    const s = stats[key];
    const rideId = String(r.rideId || '');

    if (rideId && !s.rideIds[rideId]) {
      s.rideIds[rideId] = true;
      s.total++;

      if (String(r.status || '').toUpperCase() === 'COMPLETED') s.completed++;
      if (String(r.status || '').toUpperCase() === 'CANCELLED') s.cancelled++;
    }

    const createdAt = String(r.createdAt || '');
    if (createdAt > s.lastRideAt) s.lastRideAt = createdAt;

    const rideValue = Number(r.finalPrice || r.estimatedPrice || 0);
    s.totalValue += rideValue;

    if (String(r.status || '').toUpperCase() === 'COMPLETED') {
      s.paid += rideValue;
    }

    s.outstanding += 0;
  });

  return (customers || []).map(function(c){
    const key = customerPhoneKeyV29_(c.mobile);
    const s = stats[key];

    c.mobile = customerDisplayPhoneV29_(c.mobile);

    if (s) {
      c.totalRides = s.total;
      c.completedRides = s.completed;
      c.cancelledRides = s.cancelled;
      c.lastRideAt = s.lastRideAt || c.lastRideAt;
      c.totalValue = roundMoneyV28_(s.totalValue);
      c.customerPaidTotal = roundMoneyV28_(s.paid);
      c.customerOutstanding = roundMoneyV28_(s.outstanding);
    } else {
      c.totalValue = 0;
      c.customerPaidTotal = 0;
      c.customerOutstanding = 0;
    }

    return c;
  });
}

function rebuildCustomersFromRides(token) {
  const user = assertControlRole_(token,['OWNER','OPERATIONS']);
  ensureControlV2Ready_();
  ensureCustomerPhoneIntegrityV29_();

  const rides = getAdminRidesLegacyForV2_();
  const sh = getDb_().getSheetByName(APP.SHEETS.CUSTOMERS);
  const vals = sh.getDataRange().getValues();

  const aggregate = Object.create(null);

  rides.forEach(function(r){
    const key = customerPhoneKeyV29_(r.mobile);
    if (!key) return;

    if (!aggregate[key]) {
      aggregate[key] = {
        name:String(r.customerName || ''),
        mobile:customerDisplayPhoneV29_(r.mobile),
        total:0,
        completed:0,
        cancelled:0,
        lastRideAt:String(r.createdAt || ''),
        rideIds:Object.create(null)
      };
    }

    const a = aggregate[key];
    const rideId = String(r.rideId || '');

    if (rideId && !a.rideIds[rideId]) {
      a.rideIds[rideId] = true;
      a.total++;

      if (String(r.status || '').toUpperCase() === 'COMPLETED') a.completed++;
      if (String(r.status || '').toUpperCase() === 'CANCELLED') a.cancelled++;
    }

    const createdAt = String(r.createdAt || '');
    if (createdAt >= a.lastRideAt) {
      a.lastRideAt = createdAt;
      if (r.customerName) a.name = String(r.customerName);
    }
  });

  const existingKeys = Object.create(null);

  for (let i=1;i<vals.length;i++) {
    if (!vals[i][0]) continue;

    const row = i + 1;
    const key = customerPhoneKeyV29_(vals[i][2]);
    if (!key) continue;

    existingKeys[key] = true;
    const a = aggregate[key];

    sh.getRange(row,3).setNumberFormat('@')
      .setValue(customerDisplayPhoneV29_(vals[i][2]));

    if (a) {
      if (a.name) sh.getRange(row,2).setValue(a.name);
      sh.getRange(row,5).setValue(a.lastRideAt || vals[i][4] || '');
      sh.getRange(row,6).setValue(a.total);
      sh.getRange(row,7).setValue(a.completed);
      sh.getRange(row,8).setValue(a.cancelled);
    } else {
      sh.getRange(row,6).setValue(0);
      sh.getRange(row,7).setValue(0);
      sh.getRange(row,8).setValue(0);
    }
  }

  Object.keys(aggregate).forEach(function(key){
    if (existingKeys[key]) return;

    const a = aggregate[key];
    const id = nextSimpleIdV2_('C',sh);

    sh.appendRow([
      id,
      a.name || '',
      '',
      new Date(),
      a.lastRideAt || new Date(),
      a.total,
      a.completed,
      a.cancelled,
      '',
      true
    ]);

    const row = sh.getLastRow();
    sh.getRange(row,3).setNumberFormat('@').setValue(a.mobile);
  });

  logControlAction_(
    user,
    'REBUILD_CUSTOMERS',
    'CUSTOMER',
    '',
    'Accurate rebuild from rides without increment duplication'
  );

  invalidateControlSnapshotV23_();

  return {
    success:true,
    count:sh.getLastRow()-1
  };
}

function getDriverCommissionAccountV51_(driverId) {
  driverId = clean_(driverId);

  if (!driverId) {
    return {
      driverId:'',
      commissionDue:0,
      commissionPaid:0,
      commissionOutstanding:0
    };
  }

  const rides = getAdminRidesLegacyForV2_();
  const tx = getTransactionsRawV2_();

  let commissionDue = 0;
  let commissionPaid = 0;

  rides.forEach(function(r){
    if (
      String(r.assignedDriverId || '') === String(driverId) &&
      String(r.status || '').toUpperCase() === 'COMPLETED'
    ) {
      commissionDue += Number(r.mashwarakCommission || 0);
    }
  });

  tx.forEach(function(t){
    if (
      t.status === 'CONFIRMED' &&
      t.type === 'DRIVER_COMMISSION_PAYMENT' &&
      t.direction === 'IN' &&
      String(t.driverId || '') === String(driverId)
    ) {
      commissionPaid += Number(t.amount || 0);
    }
  });

  commissionDue = roundMoneyV28_(commissionDue);
  commissionPaid = roundMoneyV28_(commissionPaid);

  return {
    driverId:String(driverId),
    commissionDue:commissionDue,
    commissionPaid:commissionPaid,
    commissionOutstanding:roundMoneyV28_(
      Math.max(0,commissionDue-commissionPaid)
    )
  };
}

function calculateAllDriverCommissionAccountsV51_(rides, tx) {
  const map = Object.create(null);

  (rides || []).forEach(function(r){
    const driverId = String(r.assignedDriverId || '');
    if (!driverId) return;

    if (!map[driverId]) {
      map[driverId] = {
        commissionDue:0,
        commissionPaid:0,
        commissionOutstanding:0
      };
    }

    if (String(r.status || '').toUpperCase() === 'COMPLETED') {
      map[driverId].commissionDue += Number(r.mashwarakCommission || 0);
    }
  });

  (tx || []).forEach(function(t){
    if (
      t.status !== 'CONFIRMED' ||
      t.type !== 'DRIVER_COMMISSION_PAYMENT' ||
      t.direction !== 'IN' ||
      !t.driverId
    ) return;

    const driverId = String(t.driverId);

    if (!map[driverId]) {
      map[driverId] = {
        commissionDue:0,
        commissionPaid:0,
        commissionOutstanding:0
      };
    }

    map[driverId].commissionPaid += Number(t.amount || 0);
  });

  Object.keys(map).forEach(function(driverId){
    const x = map[driverId];

    x.commissionDue = roundMoneyV28_(x.commissionDue);
    x.commissionPaid = roundMoneyV28_(x.commissionPaid);
    x.commissionOutstanding = roundMoneyV28_(
      Math.max(0,x.commissionDue-x.commissionPaid)
    );
  });

  return map;
}

function addControlTransaction(token, payload) {
  const user = assertControlRole_(token,['OWNER','ACCOUNTS']);
  ensureControlV2Ready_();
  payload = payload || {};

  const type = String(payload.type || '').toUpperCase();
  const allowedTypes = ['CUSTOMER_PAYMENT','DRIVER_PAYMENT','DRIVER_COMMISSION_PAYMENT','EXPENSE','REFUND'];
  if (allowedTypes.indexOf(type)===-1) throw new Error('نوع الحركة غير صحيح.');

  const amount = Number(payload.amount || 0);
  if (!(amount > 0)) throw new Error('أدخل مبلغ صحيح.');

  let direction = String(payload.direction || '').toUpperCase();
  if (!direction) {
    direction = (
      type === 'CUSTOMER_PAYMENT' ||
      type === 'DRIVER_COMMISSION_PAYMENT'
    ) ? 'IN' : 'OUT';
  }
  if (['IN','OUT'].indexOf(direction)===-1) throw new Error('اتجاه الحركة غير صحيح.');

  if (type === 'DRIVER_COMMISSION_PAYMENT') {
    const driverId = clean_(payload.driverId);
    if (!driverId) throw new Error('اختر الكابتن الذي سدد العمولة.');

    direction = 'IN';

    const account = getDriverCommissionAccountV51_(driverId);

    if (!(account.commissionOutstanding > 0)) {
      throw new Error('لا توجد عمولة مستحقة على هذا الكابتن.');
    }

    if (amount > account.commissionOutstanding + 0.009) {
      throw new Error(
        'المبلغ أكبر من العمولة المتبقية على الكابتن وهي '
        + account.commissionOutstanding
        + ' ج.م'
      );
    }
  }

  const method = String(payload.paymentMethod || 'CASH').toUpperCase();
  const status = String(payload.status || 'CONFIRMED').toUpperCase();
  if (['CONFIRMED','PENDING','CANCELLED'].indexOf(status)===-1) throw new Error('حالة الحركة غير صحيحة.');

  const sh = getDb_().getSheetByName(APP.SHEETS.TRANSACTIONS);
  const id = nextSimpleIdV2_('T',sh);
  const createdAt = new Date();
  const rideId = clean_(payload.rideId);
  const customerMobile = normalizePhone_(payload.customerMobile);
  const driverId = clean_(payload.driverId);
  const reference = clean_(payload.reference);
  const notes = clean_(payload.notes);

  sh.appendRow([
    id,createdAt,rideId,customerMobile,
    driverId,type,direction,amount,method,status,
    reference,notes,user.userId,user.name
  ]);

  logControlAction_(user,'CREATE_TRANSACTION','TRANSACTION',id,JSON.stringify({
    type:type,amount:amount,rideId:rideId
  }));

  invalidateControlSnapshotV23_();

  return {
    success:true,
    transactionId:id,
    transaction:{
      transactionId:id,
      createdAt:fmtDateTime_(createdAt),
      createdDate:Utilities.formatDate(createdAt,APP.TZ,'yyyy-MM-dd'),
      rideId:rideId,
      customerMobile:customerMobile,
      driverId:driverId,
      type:type,
      direction:direction,
      amount:amount,
      paymentMethod:method,
      status:status,
      reference:reference,
      notes:notes,
      createdByUserId:user.userId,
      createdByName:user.name
    }
  };
}

function saveControlUser(token, payload) {
  const actor = assertControlRole_(token,['OWNER']);
  ensureControlV2Ready_();
  payload = payload || {};

  const name = clean_(payload.name);
  const username = clean_(payload.username).toLowerCase();
  const role = String(payload.role || 'VIEWER').toUpperCase();
  const active = payload.active !== false && String(payload.active).toUpperCase() !== 'FALSE';
  const password = String(payload.password || '');
  const allowedRoles = ['OWNER','OPERATIONS','ACCOUNTS','REVIEWER','VIEWER'];

  if (!name) throw new Error('اكتب اسم المستخدم.');
  if (!/^[a-zA-Z0-9._-]{3,30}$/.test(username)) throw new Error('اسم الدخول غير صحيح.');
  if (allowedRoles.indexOf(role)===-1) throw new Error('الصلاحية غير صحيحة.');

  const sh = getDb_().getSheetByName(APP.SHEETS.ADMIN_USERS);
  const vals = sh.getDataRange().getValues();
  let row=0;

  for (let i=1;i<vals.length;i++) {
    if (String(vals[i][0])===String(payload.userId || '')) row=i+1;
    if (!row && String(vals[i][2] || '').toLowerCase()===username && String(vals[i][0])!==String(payload.userId || '')) {
      throw new Error('اسم الدخول مستخدم بالفعل.');
    }

    sh.getRange(row,3).setValue(username);

      if (password.length<6) throw new Error('كلمة المرور يجب ألا تقل عن 6 أحرف.');
      sh.getRange(row,4).setValue(hash_(password));
    }
    sh.getRange(row,5).setValue(role);
    sh.getRange(row,6).setValue(active);
    sh.getRange(row,9).setValue(clean_(payload.phone));
    sh.getRange(row,10).setValue(clean_(payload.notes));
    logControlAction_(actor,'UPDATE_USER','USER',String(payload.userId),username);
    return {success:true,userId:String(payload.userId),updated:true};
  }

  if (password.length<6) throw new Error('كلمة المرور يجب ألا تقل عن 6 أحرف.');

function saveControlRideOpsV25(token, payload) {
  const user = assertControlRole_(token,['OWNER','OPERATIONS']);
  ensureRideOpsV25_();

  payload = payload || {};

  const rideId = clean_(payload.rideId);
  if (!rideId) throw new Error('رقم المشوار غير موجود.');

  const allowedStatuses = [
    'NEW','CONTACTED','CONFIRMED','ASSIGNED',
    'IN_PROGRESS','COMPLETED','CANCELLED'
  ];

  const status = String(payload.status || 'NEW').toUpperCase();
  if (allowedStatuses.indexOf(status) === -1) {
    throw new Error('حالة المشوار غير صحيحة.');
  }

  const finalPriceText = String(
    payload.finalPrice === null || payload.finalPrice === undefined
      ? ''
      : payload.finalPrice
  ).trim();

  const finalPrice = finalPriceText === '' ? 0 : Number(finalPriceText);

  if (!isFinite(finalPrice) || finalPrice < 0) {
    throw new Error('السعر يجب ألا يكون أقل من صفر.');
  }

  const sh = getDb_().getSheetByName(APP.SHEETS.RIDES);
  const row = findRowById_(sh,rideId);
  if (!row) throw new Error('المشوار غير موجود.');

  const estimatedPrice = estimatedCol
    ? Number(sh.getRange(row,estimatedCol).getValue() || 0)
    : 0;

  // Manual final price wins. If it is still blank, the application's
  // stored estimated price is the source amount.
  const effectivePrice = finalPrice > 0 ? finalPrice : estimatedPrice;
  const finance = calculateRideFinanceV28_(effectivePrice);

  let driverId = clean_(payload.assignedDriverId);
  let driverName = '';

  if (driverId) {
    const drivers = getAdminDriversLegacyForV2_();
    const driver = drivers.find(function(d){
      return String(d.driverId) === String(driverId);
    });

    if (!driver) throw new Error('الكابتن المحدد غير موجود.');
    if (driver.adminStatus !== 'APPROVED') {
      throw new Error('لا يمكن تعيين كابتن غير معتمد.');
    }

    driverName = driver.fullName;
  }

  rideSetV25_(sh,row,map,'Final Price',finalPrice > 0 ? finalPrice : '');
  rideSetV25_(sh,row,map,'Driver Cost',finance.price > 0 ? finance.driverDue : '');
  rideSetV25_(sh,row,map,'Mashwarak Commission',finance.price > 0 ? finance.commission : '');

  rideSetV25_(sh,row,map,'Assigned Driver ID',driverId);
  rideSetV25_(sh,row,map,'Assigned Driver Name',driverName);
  rideSetV25_(sh,row,map,'Status',status);
  rideSetV25_(sh,row,map,'Admin Notes',clean_(payload.adminNotes));
  rideSetV25_(sh,row,map,'Updated At',new Date());

  logControlAction_(
    user,
    'UPDATE_RIDE_OPS',
    'RIDE',
    rideId,
    JSON.stringify({
      status:status,
      driverId:driverId,
      finalPrice:finalPrice,
      effectivePrice:finance.price,
      commissionRate:finance.commissionRate,
      mashwarakCommission:finance.commission,
      driverCost:finance.driverDue
    })
  );

  invalidateControlSnapshotV23_();

  return {
    success:true,
    rideId:rideId,
    status:status,
    assignedDriverId:driverId,
    assignedDriverName:driverName,
    effectivePrice:finance.price,
    commissionRate:finance.commissionRate,

function ensureRideCommissionV28_() {

  const sh = getDb_().getSheetByName(APP.SHEETS.RIDES);

  ]);

  if (sh.getLastRow() >= 2) {

    const finalIndex = Number(map['final price'] || 0) - 1;

function ensureRideWorkflowV50_() {
  const props = PropertiesService.getScriptProperties();
  if (props.getProperty('MASHWARAK_RIDE_WORKFLOW_V50_READY') === '1') return;

  const sh = getDb_().getSheetByName(APP.SHEETS.RIDES);

  ensureNamedHeaders_(sh, [
    'Workflow Stage','Contacted At','Confirmed At','Assigned At',
    'Started At','Arrived At','Completed At','Cancelled At','Cancel Reason'
  ]);

  if (sh.getLastRow() >= 2) {
    const map = rideHeaderMapV25_(sh);
    const stageCol = map['workflow stage'];
    const statusCol = map['status'];
    const arrivedCol = map['arrived at'];
    const rows = sh.getRange(2,1,sh.getLastRow()-1,sh.getLastColumn()).getValues();
    const stageValues = [];

    rows.forEach(function(r){
      const existing = stageCol ? String(r[stageCol-1] || '').toUpperCase() : '';
      if (existing) {
        stageValues.push([existing]);
        return;
      }

      const status = statusCol ? String(r[statusCol-1] || 'NEW').toUpperCase() : 'NEW';
      const arrived = arrivedCol ? r[arrivedCol-1] : '';

      let stage = status;
      if (status === 'IN_PROGRESS' && arrived) stage = 'SETTLEMENT';
      if (['NEW','CONTACTED','CONFIRMED','ASSIGNED','IN_PROGRESS','COMPLETED','CANCELLED'].indexOf(stage) === -1) {
        stage = 'NEW';
      }

      stageValues.push([stage]);
    });

    if (stageValues.length) {
      sh.getRange(2,stageCol,stageValues.length,1).setValues(stageValues);
    }
  }

  props.setProperty('MASHWARAK_RIDE_WORKFLOW_V50_READY','1');
}

function rideGetV50_(sh,row,map,header,fallback) {
  const col = map[String(header).toLowerCase()];
  if (!col) return fallback;
  const value = sh.getRange(row,col).getValue();
  return (value === null || typeof value === 'undefined' || value === '') ? fallback : value;
}

function deriveRideWorkflowStageV50_(status, storedStage, arrivedAt) {
  const stage = String(storedStage || '').toUpperCase();
  const valid = [
    'NEW','CONTACTED','CONFIRMED','ASSIGNED',
    'IN_PROGRESS','SETTLEMENT','COMPLETED','CANCELLED'
  ];

  if (valid.indexOf(stage) !== -1) return stage;

  const s = String(status || 'NEW').toUpperCase();
  if (s === 'IN_PROGRESS' && arrivedAt) return 'SETTLEMENT';
  if (valid.indexOf(s) !== -1) return s;
  return 'NEW';
}

function rideWorkflowStatusForStageV50_(stage) {
  stage = String(stage || '').toUpperCase();

  if (stage === 'SETTLEMENT') return 'IN_PROGRESS';

  if ([
    'NEW','CONTACTED','CONFIRMED','ASSIGNED',
    'IN_PROGRESS','COMPLETED','CANCELLED'
  ].indexOf(stage) !== -1) {
    return stage;
  }

  throw new Error('مرحلة التشغيل غير صحيحة.');
}

function getRideFinancialStateV50_(rideId, sh, row, map) {
  const estimatedPrice = Number(rideGetV50_(sh,row,map,'Estimated Price',0) || 0);
  const finalPrice = Number(rideGetV50_(sh,row,map,'Final Price',0) || 0);
  const price = finalPrice > 0 ? finalPrice : estimatedPrice;
  const finance = calculateRideFinanceV28_(price);

  let customerPaid = 0;
  let driverPaid = 0;
  let expenses = 0;

  const tx = getTransactionsRawV2_();

  tx.forEach(function(t){
    if (String(t.rideId || '') !== String(rideId)) return;
    if (String(t.status || '').toUpperCase() !== 'CONFIRMED') return;

    if (t.type === 'CUSTOMER_PAYMENT' && t.direction === 'IN') {
      customerPaid += Number(t.amount || 0);
    } else if (t.type === 'DRIVER_PAYMENT' && t.direction === 'OUT') {
      driverPaid += Number(t.amount || 0);
    } else if (t.type === 'EXPENSE' && t.direction === 'OUT') {
      expenses += Number(t.amount || 0);
    }
  });

  return {
    price:roundMoneyV28_(price),
    commission:finance.commission,
    driverDue:finance.driverDue,
    customerPaid:roundMoneyV28_(customerPaid),
    customerBalance:roundMoneyV28_(Math.max(0,price-customerPaid)),
    driverPaid:roundMoneyV28_(driverPaid),
    driverBalance:roundMoneyV28_(Math.max(0,finance.driverDue-driverPaid)),
    expenses:roundMoneyV28_(expenses)
  };
}

function advanceRideWorkflowV50(token, payload) {
  const started = new Date().getTime();

  ensureRideOpsV25_();

  if (!rideId) throw new Error('رقم المشوار غير موجود.');

  const row = findRowById_(sh,rideId);

  const rowValues = sh.getRange(row,1,1,sh.getLastColumn()).getValues()[0];

  const currentStage = deriveRideWorkflowStageV50_(
    status,
    value('Workflow Stage',''),
    value('Arrived At','')
  );

  if (currentStage === 'COMPLETED') {
    throw new Error('المشوار مغلق بالفعل.');
  }
  if (currentStage === 'CANCELLED') {
    throw new Error('المشوار ملغي.');
  }

  let nextStage = currentStage;
  let actionName = '';
  let detail = '';
  let driverCommissionEarned = 0;

  const sheetPatch = {'Updated At':now};
  const ridePatch = {updatedAt:fmtDateTime_(now)};

  if (action === 'CONTACT') {
    if (currentStage !== 'NEW') throw new Error('هذه الخطوة تمت بالفعل.');

    nextStage = 'CONTACTED';

    sheetPatch['Status'] = 'CONTACTED';
    sheetPatch['Workflow Stage'] = nextStage;
    sheetPatch['Contacted At'] = now;

    ridePatch.status = 'CONTACTED';
    ridePatch.workflowStage = nextStage;

    detail = 'بدأ التواصل مع العميل';

      ? priceInput

      'Mashwarak Commission':finance.commission,

      finalPrice:finance.price,
      driverCost:finance.driverDue,
      captainNet:finance.driverDue,
      mashwarakCommission:finance.commission,
      commissionRate:finance.commissionRate,
      adminNotes:notes,
      status:'CONFIRMED',
      workflowStage:'CONFIRMED',
      confirmedAt:fmtDateTime_(now),
      netProfit:roundMoneyV28_(
        finance.commission - Number(value('Ride Expenses',0) || 0)
      )
    });

    nextStage = 'CONFIRMED';
    actionName = 'RIDE_CONFIRMED';
    detail = 'العميل أكد المشوار | السعر ' + finance.price + ' ج.م';

      assignedDriverId:driver.driverId,
      assignedDriverName:driver.fullName,
      status:'ASSIGNED',
      workflowStage:'ASSIGNED',
      assignedAt:fmtDateTime_(now)

    }

    const driverId = String(value('Assigned Driver ID','') || '');

    if (!driverId) {
      throw new Error('لا يمكن إغلاق المشوار بدون كابتن معيّن.');

    const reason = clean_(payload.reason);

      'Status':'CANCELLED',

function cacheJsonV23_(key, ttlSeconds, builder) {
  const cache = CacheService.getScriptCache();

  try {
    const raw = cache.get(key);
    if (raw) return JSON.parse(raw);
  } catch(e) {}

  const value = builder();
  try {
    const encoded = JSON.stringify(value);
    // Apps Script CacheService item limit is finite.
    // Cache only reasonably sized datasets; otherwise return normally.
    if (encoded.length < 90000) {
      cache.put(key, encoded, ttlSeconds);
    }
  } catch(e) {}

  return value;
}

function enrichControlDataV23_(rides, drivers, tx) {
  const expenseByRide = Object.create(null);
  const tripCounts = Object.create(null);
  const completedCounts = Object.create(null);
  const accounts = calculateAllDriverCommissionAccountsV51_(rides,tx);

  tx.forEach(function(t){
    if (t.status !== 'CONFIRMED') return;

    if (
      t.type === 'EXPENSE' &&
      t.direction === 'OUT' &&
      t.rideId
    ) {
      expenseByRide[t.rideId] =
        (expenseByRide[t.rideId] || 0) + Number(t.amount || 0);
    }
  });

  rides.forEach(function(r){
    const price = Number(r.finalPrice || r.estimatedPrice || 0);
    const finance = calculateRideFinanceV28_(price);

    r.driverCost = finance.driverDue;
    r.captainNet = finance.driverDue;
    r.mashwarakCommission = finance.commission;
    r.commissionRate = finance.commissionRate;

    r.customerPaymentMode = 'DIRECT_TO_DRIVER';
    r.customerPaid = 0;
    r.customerBalance = 0;
    r.driverPaid = 0;
    r.driverBalance = 0;
    r.paymentStatus = 'DIRECT_TO_DRIVER';
    r.paymentMethods = [];
    r.paymentMethodLabel = 'العميل يدفع للكابتن مباشرة';

    r.rideCommissionDue = r.status === 'COMPLETED'
      ? finance.commission
      : 0;

    r.rideExpenses = Number(expenseByRide[r.rideId] || 0);
    r.netProfit = roundMoneyV28_(finance.commission-r.rideExpenses);

    if (r.assignedDriverId) {
      tripCounts[r.assignedDriverId] =
        (tripCounts[r.assignedDriverId] || 0) + 1;

      if (r.status === 'COMPLETED') {
        completedCounts[r.assignedDriverId] =
          (completedCounts[r.assignedDriverId] || 0) + 1;
      }
    }
  });

  drivers.forEach(function(d){
    const acc = accounts[d.driverId] || {
      commissionDue:0,
      commissionPaid:0,
      commissionOutstanding:0
    };

    d.totalTrips = Number(tripCounts[d.driverId] || 0);
    d.completedTrips = Number(completedCounts[d.driverId] || 0);
    d.commissionDue = Number(acc.commissionDue || 0);
    d.commissionPaid = Number(acc.commissionPaid || 0);
    d.commissionOutstanding = Number(acc.commissionOutstanding || 0);
    d.paidToDriver = 0;
  });

  return {rides:rides,drivers:drivers};
}

function buildControlDashboardV23_(rides, drivers, tx) {
  const today = Utilities.formatDate(new Date(), APP.TZ, 'yyyy-MM-dd');

  const counts = {
    NEW:0,
    CONTACTED:0,
    CONFIRMED:0,
    ASSIGNED:0,
    IN_PROGRESS:0,
    COMPLETED:0,
    CANCELLED:0
  };

  let commissionCollectedToday = 0;
  let completedCommissionTotal = 0;
  let expensesToday = 0;
  let legacyDriverPaidToday = 0;
  let driverCommissionOutstanding = 0;
  let driversUnderReview = 0;
  let driversApproved = 0;

  rides.forEach(function(r){
    if (Object.prototype.hasOwnProperty.call(counts,r.status)) counts[r.status]++;

    if (r.status === 'COMPLETED') {
      completedCommissionTotal += Number(r.mashwarakCommission || 0);
    }
  });

  drivers.forEach(function(d){
    if (d.adminStatus === 'UNDER_REVIEW') driversUnderReview++;
    if (d.adminStatus === 'APPROVED') driversApproved++;

    driverCommissionOutstanding += Number(
      d.commissionOutstanding || 0
    );
  });

  tx.forEach(function(t){
    if (t.status !== 'CONFIRMED' || t.createdDate !== today) return;

    if (
      t.type === 'DRIVER_COMMISSION_PAYMENT' &&
      t.direction === 'IN'
    ) {
      commissionCollectedToday += Number(t.amount || 0);
    } else if (
      t.type === 'EXPENSE' &&
      t.direction === 'OUT'
    ) {
      expensesToday += Number(t.amount || 0);
    } else if (
      t.type === 'DRIVER_PAYMENT' &&
      t.direction === 'OUT'
    ) {
      legacyDriverPaidToday += Number(t.amount || 0);
    }
  });

  return {
    ridesTotal:rides.length,
    newRides:counts.NEW,
    contactedRides:counts.CONTACTED,
    confirmedRides:counts.CONFIRMED,
    assignedRides:counts.ASSIGNED,
    inProgressRides:counts.IN_PROGRESS,
    completedRides:counts.COMPLETED,
    cancelledRides:counts.CANCELLED,

    driversTotal:drivers.length,
    driversUnderReview:driversUnderReview,
    driversApproved:driversApproved,

    revenueToday:roundMoneyV28_(commissionCollectedToday),
    commissionCollectedToday:roundMoneyV28_(commissionCollectedToday),
    completedCommissionTotal:roundMoneyV28_(completedCommissionTotal),
    driverCommissionOutstanding:roundMoneyV28_(driverCommissionOutstanding),

    expensesToday:roundMoneyV28_(expensesToday),
    driverPaidToday:roundMoneyV28_(legacyDriverPaidToday),
    netCashToday:roundMoneyV28_(
      commissionCollectedToday-expensesToday-legacyDriverPaidToday
    ),

    customerOutstanding:0
  };
}

      cleared:cleared,
      backupName:backupFile.getName(),
      backupUrl:backupFile.getUrl(),
      docsBackupUrl:docsBackup.url || '',
      freshDocsFolderUrl:freshDocsFolder.getUrl(),
      completedAt:Utilities.formatDate(
        new Date(),
        APP.TZ,
        'yyyy-MM-dd HH:mm:ss'
      )
    };
  } finally {
    lock.releaseLock();
  }
}

function sendMashwarakAdminPush_(title, message, eventType, entityId) {
  const props=PropertiesService.getScriptProperties();
  const projectId=props.getProperty('MASHWARAK_FCM_PROJECT_ID');
  if (!projectId) return {success:false,skipped:true,error:'Firebase not configured'};

  const accessToken=getMashwarakFcmAccessToken_();
  const url='https://fcm.googleapis.com/v1/projects/' + encodeURIComponent(projectId) + '/messages:send';

  const payload={
    message:{
      topic:'mashwarak_admin_all',
      data:{
        title:String(title || 'مشوارك للإدارة'),
        message:String(message || ''),
        eventType:String(eventType || ''),
        entityId:String(entityId || '')
      },
      android:{
        priority:'HIGH'
      }
    }
  };

  const res=UrlFetchApp.fetch(url,{
    method:'post',
    contentType:'application/json',
    headers:{Authorization:'Bearer ' + accessToken},
    payload:JSON.stringify(payload),
    muteHttpExceptions:true
  });

  const status=res.getResponseCode();
  if (status<200 || status>=300) {
    throw new Error('FCM Admin ' + status + ': ' + res.getContentText());
  }

  return {success:true};
}
