name: Publish Mashwarak APK

on:
  workflow_dispatch:
    inputs:
      tag:
        description: "Release tag, example v1.5"
        required: true
        default: "v1.11.5"

permissions:
  contents: write

jobs:
  release:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Setup Java 17
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - name: Setup Gradle
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.10.2'

      - name: Restore release keystore
        shell: bash
        run: |
          printf '%s' "${{ secrets.MASHWARAK_KEYSTORE_BASE64 }}" | base64 --decode > "$RUNNER_TEMP/mashwarak-release-key.jks"
          echo "b909765d511efd2d01b73376f279863f3ed42ec0a3bbed127867773410e9a581  $RUNNER_TEMP/mashwarak-release-key.jks" | sha256sum -c -

      - name: Build signed release APK
        env:
          MASHWARAK_KEYSTORE_PATH: ${{ runner.temp }}/mashwarak-release-key.jks
          MASHWARAK_KEYSTORE_PASSWORD: ${{ secrets.MASHWARAK_KEYSTORE_PASSWORD }}
          MASHWARAK_KEY_ALIAS: ${{ secrets.MASHWARAK_KEY_ALIAS }}
          MASHWARAK_KEY_PASSWORD: ${{ secrets.MASHWARAK_KEY_PASSWORD }}
        run: gradle :app:assembleRelease

      - name: Verify signature
        shell: bash
        run: |
          APKSIGNER="$ANDROID_HOME/build-tools/$(ls "$ANDROID_HOME/build-tools" | sort -V | tail -1)/apksigner"
          "$APKSIGNER" verify --verbose app/build/outputs/apk/release/app-release.apk

      - name: Prepare stable APK name
        run: cp app/build/outputs/apk/release/app-release.apk Mashwarak.apk

      - name: Create GitHub Release
        env:
          GH_TOKEN: ${{ github.token }}
          TAG: ${{ inputs.tag }}
        shell: bash
        run: |
          if gh release view "$TAG" >/dev/null 2>&1; then
            gh release upload "$TAG" Mashwarak.apk --clobber
          else
            gh release create "$TAG" Mashwarak.apk               --title "Mashwarak $TAG"               --notes "Signed Mashwarak Android release."
          fi
